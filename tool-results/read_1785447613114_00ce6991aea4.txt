   434→export class SdkManager {
   435→  private readonly projects = new Map<string, ProjectStructure>();
   436→  private readonly defaultRuntimeConfig: LocalRuntimeConfig = {
   437→    port: 3001,
   438→    hotReload: true,
   439→    mockPlatform: true,
   440→    logLevel: "info",
   441→    dataDir: "./.eks-local",
   442→  };
   443→
   444→  // ---- Templates --------------------------------------------------------
   445→
   446→  listTemplates(): readonly ScaffoldTemplate[] {
   447→    return TEMPLATES;
   448→  }
   449→
   450→  getTemplate(id: string): ScaffoldTemplate | undefined {
   451→    return TEMPLATE_INDEX.get(id);
   452→  }
   453→
   454→  // ---- Scaffolding ------------------------------------------------------
   455→
   456→  scaffold(input: ScaffoldInput): ScaffoldResult {
   457→    const template = TEMPLATE_INDEX.get(input.templateId);
   458→    if (!template) {
   459→      throw new ProgramError({
   460→        code: "eks.program.sdk.unknown_template",
   461→        category: "validation",
   462→        message: `Unknown template: ${input.templateId}`,
   463→        userMessage: "The requested scaffold template does not exist.",
   464→      });
   465→    }
   466→    if (!/^[a-z0-9-]+$/.test(input.slug)) {
   467→      throw new ProgramError({
   468→        code: "eks.program.sdk.invalid_slug",
   469→        category: "validation",
   470→        message: "Slug must be lowercase kebab-case.",
   471→        userMessage: "Program slug must be lowercase kebab-case.",
   472→      });
   473→    }
   474→
   475→    const manifest = this.generateManifest({
   476→      slug: input.slug,
   477→      name: input.name,
   478→      version: input.version ?? "1.0.0",
   479→      description: input.description ?? template.description,
   480→      category: input.category ?? template.defaultCategory,
   481→      developerId: input.developerId,
   482→      developerName: input.developerName,
   483→      developerEmail: input.developerEmail,
   484→      capabilities: template.defaultCapabilities,
   485→      supportedCountries: ["*"],
   486→      supportedLanguages: ["en"],
   487→      dependencies: template.defaultDependencies,
   488→      resourceLimits: template.defaultResourceLimits,
   489→      privacy: template.defaultPrivacy,
   490→      aiUsage: template.defaultAiUsage,
   491→      measurementDefinitions: template.defaultMeasurementDefinitions,
   492→    });
   493→
   494→    const files = this.generateProjectFiles(template, manifest);
   495→    const rootPath = `./${input.slug}`;
   496→    const project: ProjectStructure = {
   497→      rootPath,
   498→      files,
   499→      manifest,
   500→      templateId: template.id,
   501→    };
   502→    this.projects.set(rootPath, project);
   503→
   504→    void getEventBus().publish(
   505→      buildEvent(
   506→        "eks.program.sdk.scaffolded",
   507→        { slug: input.slug, templateId: template.id, programId: manifest.id },
   508→        {},
   509→        "domain",
   510→      ),
   511→    );
   512→
   513→    return { template, manifest, project };
   514→  }
   515→
   516→  generateManifest(input: ManifestBuilderInput): ProgramManifest {
   517→    return buildManifest(input);
   518→  }
   519→
   520→  private generateProjectFiles(
   521→    template: ScaffoldTemplate,
   522→    manifest: ProgramManifest,
   523→  ): ProjectFile[] {
   524→    const files: ProjectFile[] = [];
   525→
   526→    // manifest.json
   527→    files.push({
   528→      path: "manifest.json",
   529→      content: JSON.stringify(manifest, null, 2),
   530→    });
   531→
   532→    // src/entry.ts — real, minimal program handler
   533→    files.push({
   534→      path: "src/entry.ts",
   535→      content: this.generateEntryTs(manifest, template),
   536→    });
   537→
   538→    // src/index.ts — barrel
   539→    files.push({
   540→      path: "src/index.ts",
   541→      content: `export { handler } from "./entry";\n`,
   542→    });
   543→
   544→    // README.md
   545→    files.push({
   546→      path: "README.md",
   547→      content: this.generateReadme(manifest, template),
   548→    });
   549→
   550→    // test/contract.test.ts — real test file
   551→    files.push({
   552→      path: "test/contract.test.ts",
   553→      content: this.generateTestFile(manifest),
   554→    });
   555→
   556→    // .eksprogramrc — local runtime config
   557→    files.push({
   558→      path: ".eksprogramrc.json",
   559→      content: JSON.stringify(
   560→        {
   561→          programId: manifest.id,
   562→          slug: manifest.slug,
   563→          version: semVerToString(manifest.version),
   564→          sdkVersion: semVerToString(manifest.sdkVersion),
   565→          runtime: this.defaultRuntimeConfig,
   566→        },
   567→        null,
   568→        2,
   569→      ),
   570→    });
   571→
   572→    // tsconfig.json — minimal
   573→    files.push({
   574→      path: "tsconfig.json",
   575→      content: JSON.stringify(
   576→        {
   577→          compilerOptions: {
   578→            target: "ES2022",
   579→            module: "esnext",
   580→            moduleResolution: "bundler",
   581→            strict: true,
   582→            esModuleInterop: true,
   583→            skipLibCheck: true,
   584→            outDir: "./dist",
   585→          },
   586→          include: ["src"],
   587→        },
   588→        null,
   589→        2,
   590→      ),
   591→    });
   592→
   593→    return files;
   594→  }
   595→
   596→  private generateEntryTs(
   597→    manifest: ProgramManifest,
   598→    template: ScaffoldTemplate,
   599→  ): string {
   600→    const handlerMethods: string[] = [];
   601→
   602→    if (manifest.capabilities.some((c) => c.capability === "measurement")) {
   603→      handlerMethods.push(`  async onMeasurement(ctx, event) {
   604→    // Called when a measurement is recorded for this program.
   605→    ctx.log.info("Measurement received", { value: event.payload });
   606→  }`);
   607→    }
   608→    if (manifest.capabilities.some((c) => c.capability === "competition")) {
   609→      handlerMethods.push(`  async onCompetitionStart(ctx, event) {
   610→    // Called when a competition starts.
   611→    ctx.log.info("Competition started", { competitionId: event.competitionId });
   612→  }`);
   613→    }
   614→    if (manifest.capabilities.some((c) => c.capability === "notification")) {
   615→      handlerMethods.push(`  async onNotificationRequest(ctx, request) {
   616→    // Called to send a notification.
   617→    return { delivered: true };
   618→  }`);
   619→    }
   620→    if (manifest.aiUsage?.usesAI) {
   621→      handlerMethods.push(`  async onAiPrompt(ctx, prompt) {
   622→    // Called to process an AI prompt.
   623→    return { response: "AI response placeholder" };
   624→  }`);
   625→    }
   626→
   627→    const methods =
   628→      handlerMethods.length > 0
   629→        ? handlerMethods.join("\n\n") + "\n"
   630→        : `  async onEvent(ctx, event) {
   631→    // Default event handler.
   632→    ctx.log.info("Event received", { type: event.type });
   633→  }
   634→`;
   635→
   636→    return `/**
   637→ * ${manifest.name} — generated by @eks/program-sdk
   638→ *
   639→ * Template: ${template.id}
   640→ * Version: ${semVerToString(manifest.version)}
   641→ * SDK: ${semVerToString(manifest.sdkVersion)}
   642→ *
   643→ * This is a real, minimal program handler that the Eks-Health runtime
   644→ * can execute. Implement the lifecycle methods below.
   645→ */
   646→
   647→import type { ProgramHandler, ProgramContext, ProgramEvent } from "@eks/program-sdk";
   648→
   649→export const handler: ProgramHandler = {
   650→${methods}};
   651→
   652→export default handler;
   653→`;
   654→  }
   655→
   656→  private generateReadme(
   657→    manifest: ProgramManifest,
   658→    template: ScaffoldTemplate,
   659→  ): string {
   660→    const capList =
   661→      manifest.capabilities.length > 0
   662→        ? manifest.capabilities
   663→            .map((c) => `- **${c.capability}**: ${c.reason}`)
   664→            .join("\n")
   665→        : "- (none)";
   666→
   667→    return `# ${manifest.name}
   668→
   669→${manifest.description}
   670→
   671→- **Program ID**: \`${manifest.id}\`
   672→- **Slug**: \`${manifest.slug}\`
   673→- **Version**: ${semVerToString(manifest.version)}
   674→- **SDK Version**: ${semVerToString(manifest.sdkVersion)}
   675→- **Category**: ${manifest.category}
   676→- **Template**: ${template.name}
   677→
   678→## Capabilities
   679→
   680→${capList}
   681→
   682→## Privacy
   683→
   684→- **Data collected**: ${manifest.privacy.dataCollected.length > 0 ? manifest.privacy.dataCollected.join(", ") : "none"}

   745→
   746→  package(
   747→    projectPath: string,
   748→    manifest: ProgramManifest,
   749→    signingKey?: SigningKeyPair,
   750→  ): PackageOutput {
   751→    const project = this.projects.get(projectPath);
   752→    if (!project) {
   753→      throw new ProgramError({
   754→        code: "eks.program.sdk.project_not_found",
   755→        category: "not_found",
   756→        message: `Project not found at ${projectPath}. Call scaffold() first.`,
   757→        userMessage: "Project not found.",
   758→      });
   759→    }
   760→
   761→    const fileEntries: PackageFileEntry[] = project.files.map((f) => {
   762→      const buf = Buffer.from(f.content, "utf-8");
   763→      const sha256 = createHash("sha256").update(buf).digest("hex");
   764→      return { path: f.path, sha256, size: buf.byteLength };
   765→    });
   766→
   767→    const totalBytes = fileEntries.reduce((sum, f) => sum + f.size, 0);
   768→    const fingerprint = manifestFingerprint(manifest);
   769→
   770→    let signature: ManifestSignature | undefined;
   771→    if (signingKey) {
   772→      const signed = signManifest(manifest, signingKey, "developer");
   773→      signature = signed.signature;
   774→    }
   775→
   776→    const output: PackageOutput = {
   777→      packageId: asPackageId(generateId("pkg_")),
   778→      manifest,
   779→      fingerprint,
   780→      signature,
   781→      files: fileEntries,
   782→      packagedAt: getClock().iso(),
   783→      totalBytes,
   784→    };
   785→
   786→    void getEventBus().publish(
   787→      buildEvent(
   788→        "eks.program.sdk.packaged",
   789→        {
   790→          packageId: output.packageId,
   791→          programId: manifest.id,
   792→          fileCount: fileEntries.length,
   793→          totalBytes,
   794→          signed: !!signature,
   795→        },
   796→        {},
   797→        "domain",
   798→      ),
   799→    );
   800→
   801→    return output;
   802→  }

   809→  // ---- Contract validation ---------------------------------------------

   811→  validateContract(manifest: ProgramManifest): ContractValidationResult {
   812→    const errors: ContractIssue[] = [];
   813→    const warnings: ContractIssue[] = [];
   814→
   815→    // Capabilities must exist in the catalog
   816→    for (const c of manifest.capabilities) {
   817→      const cap = getCapability(c.capability as CapabilityId);
   818→      if (!cap) {
   819→        errors.push({
   820→          field: `capabilities.${c.capability}`,
   821→          code: "unknown_capability",
   822→          message: `Capability ${c.capability} is not in the platform catalog.`,
   823→        });
   824→      }
   825→    }
   826→
   827→    // Supported APIs must be a subset of known APIs
   828→    const knownApis = new Set(["rest", "websocket", "webhook"]);
   829→    for (const api of manifest.supportedApis) {
   830→      if (!knownApis.has(api)) {
   831→        errors.push({
   832→          field: "supportedApis",
   833→          code: "unknown_api",
   834→          message: `API ${api} is not supported by the platform.`,
   835→        });
   836→      }
   837→    }
   838→
   839→    // Platform version compatibility
   840→    const currentPlatform = parseSemVer("2.0.0");
   841→    if (compareSemVer(manifest.minPlatformVersion, currentPlatform) > 0) {
   842→      errors.push({
   843→        field: "minPlatformVersion",
   844→        code: "incompatible",
   845→        message: `Program requires platform ${semVerToString(manifest.minPlatformVersion)} but current is ${semVerToString(currentPlatform)}.`,
   846→      });
   847→    }
   848→
   849→    // SDK version
   850→    const minSdk = parseSemVer("1.0.0");
   851→    if (compareSemVer(manifest.sdkVersion, minSdk) < 0) {
   852→      warnings.push({
   853→        field: "sdkVersion",
   854→        code: "prerelease_sdk",
   855→        message: `SDK version ${semVerToString(manifest.sdkVersion)} is pre-release.`,
   856→      });
   857→    }
   858→
   859→    // Dependency ranges must be parseable
   860→    for (const d of manifest.dependencies) {
   861→      if (!isValidRangeSyntax(d.versionRange)) {
   862→        errors.push({
   863→          field: `dependencies.${d.name}`,
   864→          code: "invalid_range",
   865→          message: `Invalid version range: ${d.versionRange}`,
   866→        });
   867→      }
   868→    }
   869→
   870→    return { valid: errors.length === 0, errors, warnings };
   871→  }

   875→  generateDocs(manifest: ProgramManifest): string {
   876→    const lines: string[] = [];
   877→    lines.push(`# ${manifest.name}`);
   878→    lines.push("");
   879→    lines.push(`> ${manifest.description}`);
   880→    lines.push("");
   881→    lines.push("## Identity");
   882→    lines.push("");
   883→    lines.push(`| Field | Value |`);
   884→    lines.push(`| --- | --- |`);
   885→    lines.push(`| Program ID | \`${manifest.id}\` |`);
   886→    lines.push(`| Slug | \`${manifest.slug}\` |`);
   887→    lines.push(`| Version | ${semVerToString(manifest.version)} |`);
   888→    lines.push(`| SDK Version | ${semVerToString(manifest.sdkVersion)} |`);
   889→    lines.push(`| Category | ${manifest.category} |`);
   890→    lines.push(`| Kind | ${manifest.kind} |`);
   891→    lines.push(`| Developer | ${manifest.developer.name} <${manifest.developer.email}> |`);
   892→    lines.push("");
   893→
   894→    lines.push("## Capabilities");
   895→    lines.push("");
   896→    if (manifest.capabilities.length === 0) {
   897→      lines.push("This program requests no capabilities.");
   898→    } else {
   899→      lines.push("| Capability | Reason | Purposes | Scope |");
   900→      lines.push("| --- | --- | --- | --- |");
   901→      for (const c of manifest.capabilities) {
   902→        const cap = getCapability(c.capability as CapabilityId);
   903→        const sensitive = cap?.sensitive ? " (sensitive)" : "";
   904→        lines.push(
   905→          `| ${c.capability}${sensitive} | ${c.reason} | ${(c.purposes ?? []).join(", ") || "—"} | ${c.scope ?? "self"} |`,
   906→        );
   907→      }
   908→    }
   909→    lines.push("");
   910→
   911→    lines.push("## Permissions");
   912→    lines.push("");
   913→    if (manifest.permissions.length === 0) {
   914→      lines.push("No additional permissions requested.");
   915→    } else {
   916→      for (const p of manifest.permissions) {
   917→        lines.push(`- \`${p}\``);
   918→      }
   919→    }
   920→    lines.push("");
   921→
   922→    lines.push("## Privacy");
   923→    lines.push("");
   924→    lines.push(`- **Data collected**: ${manifest.privacy.dataCollected.length > 0 ? manifest.privacy.dataCollected.join(", ") : "none"}`);
   925→    lines.push(`- **Data usage**: ${manifest.privacy.dataUsage}`);
   926→    lines.push(`- **Retention**: ${manifest.privacy.retentionDays} days`);
   927→    lines.push(`- **Third-party sharing**: ${manifest.privacy.thirdPartySharing}`);
   928→    lines.push(`- **Anonymization applied**: ${manifest.privacy.anonymizationApplied}`);
   929→    lines.push(`- **Residency regions**: ${manifest.privacy.residencyRegions.join(", ")}`);
   930→    lines.push("");
   931→
   932→    lines.push("## AI Usage");
   933→    lines.push("");
   934→    if (manifest.aiUsage.usesAI) {
   935→      lines.push(`- **Purpose**: ${manifest.aiUsage.purpose}`);
   936→      lines.push(`- **Provider**: ${manifest.aiUsage.provider ?? "n/a"}`);
   937→      lines.push(`- **Model family**: ${manifest.aiUsage.modelFamily ?? "n/a"}`);
   938→      lines.push(`- **Training data used**: ${manifest.aiUsage.trainingDataUsed ?? false}`);
   939→      lines.push(`- **Explanation**: ${manifest.aiUsage.humanReadableExplanation ?? "n/a"}`);
   940→    } else {
   941→      lines.push("This program does not use AI.");
   942→    }
   943→    lines.push("");
   944→
   945→    lines.push("## Resources");
   946→    lines.push("");
   947→    const rl = manifest.resourceLimits;
   948→    if (Object.keys(rl).length === 0) {
   949→      lines.push("Default platform quotas apply.");
   950→    } else {
   951→      lines.push("| Resource | Limit |");
   952→      lines.push("| --- | --- |");
   953→      for (const [k, v] of Object.entries(rl)) {
   954→        lines.push(`| ${k} | ${v} |`);
   955→      }
   956→    }
   957→    lines.push("");
   958→
   959→    lines.push("## Dependencies");
   960→    lines.push("");
   961→    if (manifest.dependencies.length === 0) {
   962→      lines.push("No external dependencies.");
   963→    } else {
   964→      lines.push("| Name | Version | Type |");
   965→      lines.push("| --- | --- | --- |");
   966→      for (const d of manifest.dependencies) {
   967→        lines.push(`| ${d.name} | ${d.versionRange} | ${d.type} |`);
   968→      }
   969→    }
   970→    lines.push("");
   971→
   972→    lines.push("## Supported APIs");
   973→    lines.push("");
   974→    for (const api of manifest.supportedApis) {
   975→      lines.push(`- ${api}`);
   976→    }
   977→    lines.push("");
   978→
   979→    return lines.join("\n");
   980→  }

   982→  // ---- Upgrade simulation ----------------------------------------------
   983→
   984→  simulateUpgrade(
   985→    currentManifest: ProgramManifest,
   986→    newManifest: ProgramManifest,
   987→  ): UpgradeSimulation {
   988→    const currentCaps = new Set(currentManifest.capabilities.map((c) => c.capability));
   989→    const newCaps = new Set(newManifest.capabilities.map((c) => c.capability));
   990→
   991→    const addedCapabilities = [...newCaps].filter(
   992→      (c) => !currentCaps.has(c),
   993→    ) as CapabilityId[];
   994→    const removedCapabilities = [...currentCaps].filter(
   995→      (c) => !newCaps.has(c),
   996→    ) as CapabilityId[];
   997→
   998→    const currentPerms = new Set(currentManifest.permissions);
   999→    const newPerms = new Set(newManifest.permissions);
  1000→    const addedPermissions = [...newPerms].filter((p) => !currentPerms.has(p));
  1001→    const removedPermissions = [...currentPerms].filter((p) => !newPerms.has(p));
  1002→
  1003→    const privacyChanges: string[] = [];
  1004→    const cp = currentManifest.privacy;
  1005→    const np = newManifest.privacy;
  1006→    if (cp.dataUsage !== np.dataUsage) {
  1007→      privacyChanges.push(`dataUsage: "${cp.dataUsage}" → "${np.dataUsage}"`);
  1008→    }
  1009→    if (cp.retentionDays !== np.retentionDays) {
  1010→      privacyChanges.push(`retentionDays: ${cp.retentionDays} → ${np.retentionDays}`);
  1011→    }
  1012→    if (cp.thirdPartySharing !== np.thirdPartySharing) {
  1013→      privacyChanges.push(`thirdPartySharing: ${cp.thirdPartySharing} → ${np.thirdPartySharing}`);
  1014→    }
  1015→    if (cp.anonymizationApplied !== np.anonymizationApplied) {
  1016→      privacyChanges.push(`anonymizationApplied: ${cp.anonymizationApplied} → ${np.anonymizationApplied}`);
  1017→    }
  1018→    const addedData = np.dataCollected.filter(
  1019→      (d) => !cp.dataCollected.includes(d),
  1020→    );
  1021→    if (addedData.length > 0) {
  1022→      privacyChanges.push(`dataCollected added: ${addedData.join(", ")}`);
  1023→    }
  1024→
  1025→    const aiUsageChanges: string[] = [];
  1026→    const ca = currentManifest.aiUsage;
  1027→    const na = newManifest.aiUsage;
  1028→    if (ca.usesAI !== na.usesAI) {
  1029→      aiUsageChanges.push(`usesAI: ${ca.usesAI} → ${na.usesAI}`);
  1030→    }
  1031→    if (ca.purpose !== na.purpose) {
  1032→      aiUsageChanges.push(`purpose: "${ca.purpose ?? "—"}" → "${na.purpose ?? "—"}"`);
  1033→    }
  1034→    if (ca.provider !== na.provider) {
  1035→      aiUsageChanges.push(`provider: ${ca.provider ?? "—"} → ${na.provider ?? "—"}`);
  1036→    }
  1037→
  1038→    const resourceLimitChanges: string[] = [];
  1039→    const allResKeys = new Set([
  1040→      ...Object.keys(currentManifest.resourceLimits),
  1041→      ...Object.keys(newManifest.resourceLimits),
  1042→    ]);
  1043→    for (const k of allResKeys) {
  1044→      const cv = (currentManifest.resourceLimits as Record<string, unknown>)[k];
  1045→      const nv = (newManifest.resourceLimits as Record<string, unknown>)[k];
  1046→      if (cv !== nv) {
  1047→        resourceLimitChanges.push(`${k}: ${cv ?? "default"} → ${nv ?? "default"}`);
  1048→      }
  1049→    }
  1050→
  1051→    // Determine if re-consent is required
  1052→    const sensitiveAdded = addedCapabilities.some((c) => {
  1053→      const cap = getCapability(c as CapabilityId);
  1054→      return cap?.requiresConsent ?? false;
  1055→    });
  1056→    const requiresReconsent =
  1057→      sensitiveAdded ||
  1058→      addedPermissions.length > 0 ||
  1059→      privacyChanges.length > 0 ||
  1060→      (na.usesAI && !ca.usesAI);
  1061→
  1062→    const isBreaking =
  1063→      removedCapabilities.length > 0 ||
  1064→      removedPermissions.length > 0 ||
  1065→      compareSemVer(newManifest.version, currentManifest.version) < 0;
  1066→
  1067→    return {
  1068→      addedCapabilities,
  1069→      removedCapabilities,
  1070→      addedPermissions,
  1071→      removedPermissions,
  1072→      privacyChanges,
  1073→      aiUsageChanges,
  1074→      resourceLimitChanges,
  1075→      requiresReconsent,
  1076→      isBreaking,
  1077→    };
  1078→  }
