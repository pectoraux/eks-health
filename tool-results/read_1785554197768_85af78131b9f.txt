     1→/**
     2→ * Eks-Health Identity — Consent Engine
     3→ *
     4→ * First-class consent platform: granular, purpose-specific, field-level,
     5→ * program-specific, org-specific, time-limited, renewable, expirable,
     6→ * withdrawable, emergency-override framework, history, immutable receipts,
     7→ * versioned.
     8→ *
     9→ * No data leaves the platform without an active, versioned, receipted
    10→ * consent covering the specific purpose AND field being accessed.
    11→ * Sensitive field categories (genetics, mental_health, etc.) cannot be
    12→ * required by programs — they may only be requested as opt-in optional
    13→ * fields, and the user must explicitly approve them.
    14→ */

   168→function sha256(input: string): string {
   169→  return createHash("sha256").update(input, "utf8").digest("hex");
   170→}
   171→
   172→function computeReceiptHash(receipt: Omit<ConsentReceipt, "hash">): string {
   173→  // Deterministic serialization — keys sorted, arrays in declared order.
   174→  const payload = JSON.stringify({
   175→    id: receipt.id,
   176→    consentId: receipt.consentId,
   177→    accountId: receipt.accountId,
   178→    programId: receipt.programId,
   179→    purpose: receipt.purpose,
   180→    approvedFields: [...receipt.approvedFields].sort(),
   181→    deniedFields: [...receipt.deniedFields].sort(),
   182→    grantedAt: receipt.grantedAt,
   183→    expiresAt: receipt.expiresAt,
   184→    actor: receipt.actor,
   185→    version: receipt.version,
   186→  });
   187→  return sha256(payload);
   188→}
   189→
   190→function assertFieldsValid(label: string, fields: readonly string[]): void {
   191→  for (const f of fields) {
   192→    if (typeof f !== "string" || f.length === 0) {
   193→      throw new IdentityError({
   194→        code: "eks.identity.consent.invalid_field",
   195→        category: "validation",
   196→        message: `Invalid field in ${label}: ${String(f)}`,
   197→        userMessage: "One of the requested fields is invalid.",
   198→      });
   199→    }
   200→  }
   201→}

   207→export class ConsentManager {
   208→  private readonly consents = new Map<ConsentId, Consent>();
   209→  private readonly byAccount = new Map<AccountId, Set<ConsentId>>();
   210→  private readonly byAccountProgram = new Map<string, Set<ConsentId>>(); // `${accountId}::${programId}`
   211→  private readonly receipts = new Map<ConsentReceiptId, ConsentReceipt>();
   212→  private readonly versions = new Map<ConsentId, ConsentVersion[]>();
   213→  private readonly overrides = new Map<string, EmergencyOverride>();
   214→  private readonly overridesByAccount = new Map<AccountId, Set<string>>();
   215→
   216→  // ----- Request -----------------------------------------------------------
   217→
   218→  /**
   219→   * A program requests consent for a purpose. Creates a pending Consent
   220→   * (status="pending") and returns it so the UI can show the request to
   221→   * the user.
   222→   */
   223→  requestConsent(
   224→    accountId: AccountId,
   225→    request: ConsentPurposeRequest,
   226→    programId: string,
   227→  ): Consent {
   228→    if (!programId) {
   229→      throw new IdentityError({
   230→        code: "eks.identity.consent.invalid_program",
   231→        category: "validation",
   232→        message: "programId is required.",
   233→        userMessage: "A program id is required.",
   234→      });
   235→    }
   236→    if (!request.purpose) {
   237→      throw new IdentityError({
   238→        code: "eks.identity.consent.invalid_purpose",
   239→        category: "validation",
   240→        message: "purpose is required.",
   241→        userMessage: "A purpose is required.",
   242→      });
   243→    }
   244→    assertFieldsValid("requestedFields", [...request.requestedFields]);
   245→    if (request.optionalFields) assertFieldsValid("optionalFields", [...request.optionalFields]);
   246→    if (request.deniedFields) assertFieldsValid("deniedFields", [...request.deniedFields]);
   247→
   248→    // DENIED_BY_DEFAULT fields cannot be required — they must be optional.
   249→    const requiredSensitive = request.requestedFields.filter((f) =>
   250→      DENIED_BY_DEFAULT.has(f),
   251→    );
   252→    if (requiredSensitive.length > 0) {
   253→      throw new IdentityError({
   254→        code: "eks.identity.consent.sensitive_required",
   255→        category: "policy_violation",
   256→        message: `Sensitive fields cannot be required: ${requiredSensitive.join(", ")}. Move them to optionalFields.`,
   257→        userMessage:
   258→          "Some requested fields are sensitive and cannot be required. They must be offered as opt-in.",
   259→      });
   260→    }
   261→
   262→    const now = getClock().iso();
   263→    const consent: Consent = {
   264→      id: asConsentId(generateId("cns_")),
   265→      accountId,
   266→      programId,
   267→      purpose: request.purpose,
   268→      description: request.description,
   269→      requestedFields: [...request.requestedFields],
   270→      optionalFields: [...(request.optionalFields ?? [])],
   271→      deniedFields: [...(request.deniedFields ?? [])],
   272→      approvedFields: [],
   273→      userDeniedFields: [],
   274→      status: "pending",
   275→      version: 0,
   276→      createdAt: now,
   277→    };
   278→    this.consents.set(consent.id, consent);
   279→    this.indexConsent(consent);
   280→    this.recordVersion(consent, "requested", accountId, "Consent requested");
   281→
   282→    void getEventBus().publish(
   283→      buildEvent(
   284→        CONSENT_EVENTS.requested,
   285→        {
   286→          consentId: consent.id,
   287→          accountId,
   288→          programId,
   289→          purpose: request.purpose,
   290→          requestedFields: consent.requestedFields,
   291→          optionalFields: consent.optionalFields,
   292→        },
   293→        {},
   294→        "domain",
   295→      ),
   296→    );
   297→    return consent;
   298→  }
   299→
   300→  // ----- Grant -------------------------------------------------------------
   301→
   302→  /**
   303→   * User grants (or partially grants / denies) a pending consent request.
   304→   * Activates the consent, issues a ConsentReceipt, records version 1.
   305→   */
   306→  grant(
   307→    consentId: ConsentId,
   308→    approvedFields: readonly string[],
   309→    deniedFields: readonly string[],
   310→    durationMs: number = DEFAULT_CONSENT_DURATION_MS,
   311→    actor: AccountId | null = null,
   312→  ): Consent {
   313→    const c = this.consents.get(consentId);
   314→    if (!c) {
   315→      throw new IdentityError({
   316→        code: "eks.identity.consent.not_found",
   317→        category: "not_found",
   318→        message: `Consent ${consentId} not found.`,
   319→        userMessage: "This consent request no longer exists.",
   320→      });
   321→    }
   322→    if (c.status !== "pending") {
   323→      throw new IdentityError({
   324→        code: "eks.identity.consent.not_pending",
   325→        category: "conflict",
   326→        message: `Consent ${consentId} is in status ${c.status}, not pending.`,
   327→        userMessage: "This consent has already been resolved.",
   328→      });
   329→    }
   330→    assertFieldsValid("approvedFields", [...approvedFields]);
   331→    assertFieldsValid("deniedFields", [...deniedFields]);
   332→
   333→    // Approved fields must be a subset of requestedFields ∪ optionalFields.
   334→    const allowed = new Set<string>([...c.requestedFields, ...c.optionalFields]);
   335→    const invalidApprovals = approvedFields.filter((f) => !allowed.has(f));
   336→    if (invalidApprovals.length > 0) {
   337→      throw new IdentityError({
   338→        code: "eks.identity.consent.field_not_requestable",
   339→        category: "validation",
   340→        message: `Cannot approve fields not in the request: ${invalidApprovals.join(", ")}`,
   341→        userMessage: "Some approved fields were not part of the request.",
   342→      });
   343→    }
   344→
   345→    // The program's required fields must all be approved OR explicitly denied
   346→    // by the user (denial of a required field means the program will not get
   347→    // the data, which may make the purpose infeasible — but we let the user
   348→    // decide and surface it to the program).
   349→    const approvedSet = new Set(approvedFields);
   350→    const missingRequired = c.requestedFields.filter((f) => !approvedSet.has(f));
   351→    if (missingRequired.length > 0) {
   352→      // Auto-mark missing required fields as user-denied for transparency.
   353→      deniedFields = [...deniedFields, ...missingRequired];
   354→    }
   355→
   356→    const now = getClock().iso();
   357→    const expiresAt = new Date(Date.now() + durationMs).toISOString();
   358→    const actorId = actor ?? c.accountId;
   359→
   360→    const updated: Consent = {
   361→      ...c,
   362→      approvedFields: [...approvedFields],
   363→      userDeniedFields: [...new Set(deniedFields)],
   364→      status: "active",
   365→      version: 1,
   366→      grantedAt: now,
   367→      expiresAt,
   368→    };
   369→
   370→    // Issue receipt first so we can reference it from the consent.
   371→    const receipt = this.issueReceipt(updated, actorId);
   372→    const withReceipt: Consent = { ...updated, receiptId: receipt.id };
   373→    this.consents.set(c.id, withReceipt);
   374→    this.indexConsent(withReceipt);
   375→    this.recordVersion(withReceipt, "granted", actorId, "Consent granted by user");
   376→
   377→    void getEventBus().publish(
   378→      buildEvent(
   379→        CONSENT_EVENTS.granted,
   380→        {
   381→          consentId: withReceipt.id,
   382→          accountId: withReceipt.accountId,
   383→          programId: withReceipt.programId,
   384→          purpose: withReceipt.purpose,
   385→          approvedFields: withReceipt.approvedFields,
   386→          expiresAt,
   387→          receiptId: receipt.id,
   388→        },
   389→        {},
   390→        "domain",
   391→      ),
   392→    );
   393→    return withReceipt;
   394→  }
   395→
   396→  // ----- Revoke / Withdraw -------------------------------------------------
   397→
   398→  /**
   399→   * User withdraws consent. Future access under this consent is denied.
   400→   * The consent is retained in the history (status="withdrawn"); the
   401→   * receipt remains immutable.
   402→   */
   403→  revoke(consentId: ConsentId, reason: string, actor?: AccountId): Consent {
   404→    const c = this.consents.get(consentId);
   405→    if (!c) {
   406→      throw new IdentityError({
   407→        code: "eks.identity.consent.not_found",
   408→        category: "not_found",
   409→        message: `Consent ${consentId} not found.`,
   410→      });
   411→    }
   412→    if (c.status === "withdrawn" || c.status === "revoked") {
   413→      return c; // idempotent
   414→    }
   415→    const now = getClock().iso();
   416→    const actorId = actor ?? c.accountId;
   417→    const updated: Consent = {
   418→      ...c,
   419→      status: "withdrawn",
   420→      revokedAt: now,
   421→      revokeReason: reason,
   422→      version: c.version + 1,
   423→    };
   424→    this.consents.set(c.id, updated);
   425→    this.indexConsent(updated);
   426→    this.recordVersion(updated, "revoked", actorId, reason);
   427→
   428→    void getEventBus().publish(
   429→      buildEvent(
   430→        CONSENT_EVENTS.revoked,
   431→        {
   432→          consentId: updated.id,
   433→          accountId: updated.accountId,
   434→          programId: updated.programId,
   435→          reason,
   436→        },
   437→        {},
   438→        "domain",
   439→      ),
   440→    );
   441→    return updated;
   442→  }
   443→
   444→  // ----- Renew --------------------------------------------------------------
   445→
   446→  /**
   447→   * Extend a consent's duration. Creates a new version and a new receipt
   448→   * (the old receipt is retained as a historical record).
   449→   */
   450→  renew(consentId: ConsentId, durationMs: number, actor?: AccountId): Consent {
   451→    const c = this.consents.get(consentId);
   452→    if (!c) {
   453→      throw new IdentityError({
   454→        code: "eks.identity.consent.not_found",
   455→        category: "not_found",
   456→        message: `Consent ${consentId} not found.`,
   457→      });
   458→    }
   459→    if (c.status !== "active") {
   460→      throw new IdentityError({
   461→        code: "eks.identity.consent.not_active",
   462→        category: "conflict",
   463→        message: `Cannot renew a consent in status ${c.status}.`,
   464→        userMessage: "Only active consents can be renewed.",
   465→      });
   466→    }
   467→    const now = getClock().iso();
   468→    const actorId = actor ?? c.accountId;
   469→    const newExpiresAt = new Date(Date.now() + durationMs).toISOString();
   470→    const updated: Consent = {
   471→      ...c,
   472→      expiresAt: newExpiresAt,
   473→      version: c.version + 1,
   474→    };
   475→    // Issue a new receipt for the renewed period.
   476→    const receipt = this.issueReceipt(updated, actorId);
   477→    const withReceipt: Consent = { ...updated, receiptId: receipt.id };
   478→    this.consents.set(c.id, withReceipt);
   479→    this.indexConsent(withReceipt);
   480→    this.recordVersion(withReceipt, "renewed", actorId, `Renewed for ${durationMs}ms`);
   481→
   482→    void getEventBus().publish(
   483→      buildEvent(
   484→        CONSENT_EVENTS.granted,
   485→        {
   486→          consentId: withReceipt.id,
   487→          accountId: withReceipt.accountId,
   488→          programId: withReceipt.programId,
   489→          purpose: withReceipt.purpose,
   490→          renewed: true,
   491→          expiresAt: newExpiresAt,
   492→          receiptId: receipt.id,
   493→        },
   494→        {},
   495→        "domain",
   496→      ),
   497→    );
   498→    return withReceipt;
   499→  }
   500→
   501→  // ----- Expire -------------------------------------------------------------
   502→
   503→  /**
   504→   * Sweep expired consents. Called by the scheduler. Returns the number
   505→   * of consents transitioned to "expired".
   506→   */
   507→  expire(): number {
   508→    const now = Date.now();
   509→    let n = 0;
   510→    for (const [id, c] of this.consents) {
   511→      if (c.status !== "active") continue;
   512→      if (!c.expiresAt) continue;
   513→      if (new Date(c.expiresAt).getTime() <= now) {
   514→        const updated: Consent = {
   515→          ...c,
   516→          status: "expired",
   517→          version: c.version + 1,
   518→        };
   519→        this.consents.set(id, updated);
   520→        this.indexConsent(updated);
   521→        this.recordVersion(updated, "expired", c.accountId, "Consent expired");
   522→        void getEventBus().publish(
   523→          buildEvent(
   524→            CONSENT_EVENTS.expired,
   525→            {
   526→              consentId: id,
   527→              accountId: c.accountId,
   528→              programId: c.programId,
   529→              expiredAt: c.expiresAt,
   530→            },
   531→            {},
   532→            "domain",
   533→          ),
   534→        );
   535→        n++;
   536→      }
   537→    }
   538→    return n;
   539→  }

   541→  // ----- Query --------------------------------------------------------------
   542→
   543→  getActiveConsents(accountId: AccountId, programId?: string): readonly Consent[] {
   544→    let ids: Set<ConsentId>;
   545→    if (programId) {
   546→      ids = this.byAccountProgram.get(`${accountId}::${programId}`) ?? new Set();
   547→    } else {
   548→      ids = this.byAccount.get(accountId) ?? new Set();
   549→    }
   550→    const now = Date.now();
   551→    return [...ids]
   552→      .map((id) => this.consents.get(id)!)
   553→      .filter((c) => c && c.status === "active" && c.expiresAt && new Date(c.expiresAt).getTime() > now);
   554→  }
   555→
   556→  getConsent(id: ConsentId): Consent | undefined {
   557→    return this.consents.get(id);
   558→  }
   559→
   560→  listConsents(accountId?: AccountId): readonly Consent[] {
   561→    if (accountId) {
   562→      const ids = this.byAccount.get(accountId) ?? new Set();
   563→      return [...ids].map((id) => this.consents.get(id)!).filter(Boolean);
   564→    }
   565→    return [...this.consents.values()];
   566→  }
   567→
   568→  /**
   569→   * The real data-gateway access check.
   570→   * Returns true iff there is an active consent covering:
   571→   *   - accountId
   572→   *   - programId
   573→   *   - purpose
   574→   *   - field (field ∈ approvedFields)
   575→   * OR an active emergency override exists for the account.
   576→   */
   577→  checkAccess(
   578→    accountId: AccountId,
   579→    programId: string,
   580→    purpose: string,
   581→    field?: string,
   582→  ): boolean {
   583→    // Emergency override short-circuits (heavily audited at creation).
   584→    if (this.hasActiveOverride(accountId)) {
   585→      return true;
   586→    }
   587→    const active = this.getActiveConsents(accountId, programId);
   588→    for (const c of active) {
   589→      if (c.purpose !== purpose) continue;
   590→      if (field && !c.approvedFields.includes(field)) continue;
   591→      return true;
   592→    }
   593→    return false;
   594→  }
   595→
   596→  // ----- History / Receipts -------------------------------------------------
   597→
   598→  getHistory(accountId: AccountId): readonly ConsentVersion[] {
   599→    const ids = this.byAccount.get(accountId) ?? new Set();
   600→    const all: ConsentVersion[] = [];
   601→    for (const id of ids) {
   602→      const versions = this.versions.get(id);
   603→      if (versions) all.push(...versions);
   604→    }
   605→    return all.sort((a, b) => a.timestamp.localeCompare(b.timestamp));
   606→  }
   607→
   608→  getVersions(consentId: ConsentId): readonly ConsentVersion[] {
   609→    return this.versions.get(consentId) ?? [];
   610→  }
   611→
   612→  getReceipt(receiptId: ConsentReceiptId): ConsentReceipt | undefined {
   613→    return this.receipts.get(receiptId);
   614→  }
   615→
   616→  listReceipts(accountId?: AccountId): readonly ConsentReceipt[] {
   617→    if (!accountId) return [...this.receipts.values()];
   618→    return [...this.receipts.values()].filter((r) => r.accountId === accountId);
   619→  }

   621→  // ----- Emergency override -------------------------------------------------
   622→
   623→  /**
   624→   * Create a time-bound emergency override that bypasses consent for the
   625→   * account. Used for medical emergencies (e.g. unconscious patient). The
   626→   * override is heavily audited and expires automatically.
   627→   */
   628→  emergencyOverride(
   629→    accountId: AccountId,
   630→    reason: string,
   631→    authorizedBy: AccountId,
   632→    durationMs: number = DEFAULT_EMERGENCY_OVERRIDE_MS,
   633→  ): EmergencyOverride {
   634→    if (!reason) {
   635→      throw new IdentityError({
   636→        code: "eks.identity.consent.override_no_reason",
   637→        category: "validation",
   638→        message: "Emergency override requires a reason.",
   639→        userMessage: "A reason is required for an emergency override.",
   640→      });
   641→    }
   642→    const now = getClock().iso();
   643→    const expiresAt = new Date(Date.now() + durationMs).toISOString();
   644→    const override: EmergencyOverride = {
   645→      id: generateId("eov_"),
   646→      accountId,
   647→      reason,
   648→      authorizedBy,
   649→      createdAt: now,
   650→      expiresAt,
   651→      active: true,
   652→    };
   653→    this.overrides.set(override.id, override);
   654→    const set = this.overridesByAccount.get(accountId) ?? new Set();
   655→    set.add(override.id);
   656→    this.overridesByAccount.set(accountId, set);
   657→
   658→    void getEventBus().publish(
   659→      buildEvent(
   660→        CONSENT_EVENTS.overridden,
   661→        {
   662→          overrideId: override.id,
   663→          accountId,
   664→          authorizedBy,
   665→          reason,
   666→          expiresAt,
   667→        },
   668→        {},
   669→        "domain",
   670→      ),
   671→    );
   672→    return override;
   673→  }
   674→
   675→  revokeOverride(overrideId: string): EmergencyOverride | undefined {
   676→    const o = this.overrides.get(overrideId);
   677→    if (!o) return undefined;
   678→    const updated: EmergencyOverride = {
   679→      ...o,
   680→      active: false,
   681→      revokedAt: getClock().iso(),
   682→    };
   683→    this.overrides.set(overrideId, updated);
   684→    return updated;
   685→  }
   686→
   687→  listOverrides(accountId?: AccountId): readonly EmergencyOverride[] {
   688→    if (!accountId) return [...this.overrides.values()];
   689→    const set = this.overridesByAccount.get(accountId) ?? new Set();
   690→    return [...set].map((id) => this.overrides.get(id)!).filter(Boolean);
   691→  }
   692→
   693→  hasActiveOverride(accountId: AccountId): boolean {
   694→    const set = this.overridesByAccount.get(accountId);
   695→    if (!set) return false;
   696→    const now = Date.now();
   697→    for (const id of set) {
   698→      const o = this.overrides.get(id);
   699→      if (o && o.active && new Date(o.expiresAt).getTime() > now) return true;
   700→    }
   701→    return false;
   702→  }

   706→  private indexConsent(c: Consent): void {
   707→    let byAcct = this.byAccount.get(c.accountId);
   708→    if (!byAcct) {
   709→      byAcct = new Set();
   710→      this.byAccount.set(c.accountId, byAcct);
   711→    }
   712→    byAcct.add(c.id);
   713→    const key = `${c.accountId}::${c.programId}`;
   714→    let byAcctProg = this.byAccountProgram.get(key);
   715→    if (!byAcctProg) {
   716→      byAcctProg = new Set();
   717→      this.byAccountProgram.set(key, byAcctProg);
   718→    }
   719→    byAcctProg.add(c.id);
   720→  }

   722→  private recordVersion(
   723→    consent: Consent,
   724→    changeType: ConsentChangeType,
   725→    actor: AccountId,
   726→    reason?: string,
   727→  ): void {
   728→    const list = this.versions.get(consent.id) ?? [];
   729→    const v: ConsentVersion = {
   730→      version: consent.version,
   731→      consentId: consent.id,
   732→      changeType,
   733→      timestamp: getClock().iso(),
   734→      actor,
   735→      reason,
   736→      snapshot: consent,
   737→    };
   738→    list.push(v);
   739→    this.versions.set(consent.id, list);
   740→  }
   741→
   742→  private issueReceipt(consent: Consent, actor: AccountId): ConsentReceipt {
   743→    const id = asConsentReceiptId(generateId("rcp_"));
   744→    const partial: Omit<ConsentReceipt, "hash"> = {
   745→      id,
   746→      consentId: consent.id,
   747→      accountId: consent.accountId,
   748→      programId: consent.programId,
   749→      purpose: consent.purpose,
   750→      approvedFields: [...consent.approvedFields],
   751→      deniedFields: [...consent.userDeniedFields, ...consent.deniedFields],
   752→      grantedAt: consent.grantedAt ?? getClock().iso(),
   753→      expiresAt: consent.expiresAt ?? "",
   754→      actor,
   755→      version: consent.version,
   756→    };
   757→    const hash = computeReceiptHash(partial);
   758→    const receipt: ConsentReceipt = { ...partial, hash };
   759→    this.receipts.set(id, receipt);
   760→    return receipt;
   761→  }
