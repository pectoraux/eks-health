     1→/**
     2→ * Eks-Health Kernel — Search Architecture
     3→ *
     4→ * The platform's search abstraction layer. This module is *not* a full text
     5→ * engine — it is the contract that every search provider (Elasticsearch,
     6→ * OpenSearch, Meilisearch, Typesense, Postgres FTS, or an in-memory default)
     7→ * must implement, plus a manager that routes queries to the right index and
     8→ * a registry of well-known indices.
     9→ *
    10→ * Capabilities:
    11→ *  - Pluggable providers (default: in-memory tokenized inverted index)
    12→ *  - Named indices with declared searchable / filterable / sortable fields
    13→ *  - Full-text + phrase queries
    14→ *  - Field filters (eq, ne, in, gt, gte, lt, lte, exists, prefix)
    15→ *  - Sorting (asc/desc) on declared sortable fields
    16→ *  - Highlights (snippets with <mark> wrappers)
    17→ *  - Terms aggregations (facets for marketplace filters, etc.)
    18→ *  - Re-index on demand (rebuild from the document source)
    19→ *
    20→ * The default adapter is in-memory; production swaps in Elasticsearch/Meili.
    21→ */

   192→// ---------------------------------------------------------------------------
   193→// In-memory provider (default adapter) — tokenized inverted index
   194→// ---------------------------------------------------------------------------
   195→
   196→interface IndexedDoc {
   197→  readonly doc: SearchDocument;
   198→  /** field -> ordered tokens */
   199→  readonly fieldTokens: Map<string, string[]>;
   200→  /** token -> term frequency across all searchable fields */
   201→  readonly tf: Map<string, number>;
   202→  /** token -> set of fields containing it (for field-scoped queries) */
   203→  readonly fieldsByToken: Map<string, Set<string>>;
   204→}
   205→
   206→interface IndexState {
   207→  readonly definition: SearchIndex;
   208→  docs: Map<string, IndexedDoc>;
   209→  /** token -> Set<docId> */
   210→  inverted: Map<string, Set<string>>;
   211→  /** total token count across the index (for avgdl in BM25) */
   212→  totalTokens: number;
   213→}
   214→
   215→/** Tokenize: lowercase, split on whitespace + punctuation, drop empties. */
   216→function tokenize(text: string): string[] {
   217→  if (!text) return [];
   218→  return text
   219→    .toLowerCase()
   220→    .split(/[\s\p{P}\p{S}]+/u)
   221→    .filter((t) => t.length > 0);
   222→}
   223→
   224→function tokenizeValue(value: SearchFieldValue): string[] {
   225→  if (value === null || value === undefined) return [];
   226→  if (typeof value === "string") return tokenize(value);
   227→  if (typeof value === "number" || typeof value === "boolean") return tokenize(String(value));
   228→  if (Array.isArray(value)) {
   229→    const out: string[] = [];
   230→    for (const v of value) out.push(...tokenizeValue(v as SearchFieldValue));
   231→    return out;
   232→  }
   233→  return [];
   234→}

   236→const K1 = 1.5;
   237→const B = 0.75;
   238→
   239→export class InMemorySearchProvider implements SearchProvider {
   240→  readonly name = "in-memory";
   241→  private readonly indices = new Map<string, IndexState>();
   242→
   243→  /** Ensure an index exists in this provider. */
   244→  private ensureIndex(indexName: string): IndexState {
   245→    let state = this.indices.get(indexName);
   246→    if (!state) {
   247→      // Fall back to a permissive definition if the index wasn't pre-registered.
   248→      const def =
   249→        INDEX_DEFINITIONS[indexName as SearchIndexName] ??
   250→        ({
   251→          name: indexName,
   252→          searchableFields: [],
   253→          filterableFields: [],
   254→          sortableFields: [],
   255→        } as SearchIndex);
   256→      state = {
   257→        definition: def,
   258→        docs: new Map(),
   259→        inverted: new Map(),
   260→        totalTokens: 0,
   261→      };
   262→      this.indices.set(indexName, state);
   263→    }
   264→    return state;
   265→  }
   266→
   267→  async index(indexName: string, doc: SearchDocument): Promise<void> {
   268→    const state = this.ensureIndex(indexName);
   269→    // Remove existing version if present.
   270→    if (state.docs.has(doc.id)) {
   271→      this.removeFromInverted(state, doc.id);
   272→    }
   273→    const searchable = state.definition.searchableFields.length
   274→      ? state.definition.searchableFields
   275→      : Object.keys(doc.fields);
   276→    const tokensByField = new Map<string, string[]>();
   277→    const tf = new Map<string, number>();
   278→    const fieldsByToken = new Map<string, Set<string>>();
   279→    let docTokens = 0;
   280→    for (const field of searchable) {
   281→      const toks = tokenizeValue(doc.fields[field] ?? null);
   282→      tokensByField.set(field, toks);
   283→      docTokens += toks.length;
   284→      const seenInThisField = new Set<string>();
   285→      for (const t of toks) {
   286→        tf.set(t, (tf.get(t) ?? 0) + 1);
   287→        seenInThisField.add(t);
   288→      }
   289→      for (const t of seenInThisField) {
   290→        let set = fieldsByToken.get(t);
   291→        if (!set) {
   292→          set = new Set();
   293→          fieldsByToken.set(t, set);
   294→        }
   295→        set.add(field);
   296→      }
   297→    }
   298→    const indexed: IndexedDoc = { doc, fieldTokens: tokensByField, tf, fieldsByToken };
   299→    state.docs.set(doc.id, indexed);
   300→    for (const token of tf.keys()) {
   301→      let postings = state.inverted.get(token);
   302→      if (!postings) {
   303→        postings = new Set();
   304→        state.inverted.set(token, postings);
   305→      }
   306→      postings.add(doc.id);
   307→    }
   308→    state.totalTokens += docTokens;
   309→  }
   310→
   311→  private removeFromInverted(state: IndexState, docId: string): void {
   312→    const existing = state.docs.get(docId);
   313→    if (!existing) return;
   314→    for (const [token, set] of existing.fieldsByToken.entries()) {
   315→      void set;
   316→      const postings = state.inverted.get(token);
   317→      if (postings) {
   318→        postings.delete(docId);
   319→        if (postings.size === 0) state.inverted.delete(token);
   320→      }
   321→    }
   322→    let docLen = 0;
   323→    for (const toks of existing.fieldTokens.values()) docLen += toks.length;
   324→    state.totalTokens -= docLen;
   325→    state.docs.delete(docId);
   326→  }
   327→
   328→  async delete(indexName: string, id: string): Promise<boolean> {
   329→    const state = this.indices.get(indexName);
   330→    if (!state) return false;
   331→    if (!state.docs.has(id)) return false;
   332→    this.removeFromInverted(state, id);
   333→    return true;
   334→  }
   335→
   336→  async reindex(indexName: string): Promise<number> {
   337→    const state = this.indices.get(indexName);
   338→    if (!state) return 0;
   339→    const snapshot = [...state.docs.values()].map((d) => d.doc);
   340→    // Reset
   341→    state.docs.clear();
   342→    state.inverted.clear();
   343→    state.totalTokens = 0;
   344→    for (const doc of snapshot) {
   345→      await this.index(indexName, doc);
   346→    }
   347→    return snapshot.length;
   348→  }
   349→
   350→  async search(indexName: string, query: SearchQuery): Promise<SearchResult> {
   351→    const started = getClock().epochMs();
   352→    const state = this.ensureIndex(indexName);
   353→    const def = state.definition;
   354→    const limit = query.limit ?? 20;
   355→    const offset = query.offset ?? 0;
   356→
   357→    // --- Candidate set: start with all docs, narrow by filters ---
   358→    let candidateIds = new Set<string>(state.docs.keys());
   359→
   360→    if (query.filters && query.filters.length > 0) {
   361→      for (const f of query.filters) {
   362→        if (def.filterableFields.length && !def.filterableFields.includes(f.field)) {
   363→          throw new ValidationError(
   364→            "eks.error.search.field_not_filterable",
   365→            `Field '${f.field}' is not filterable in index '${indexName}'`,
   366→            "Search filter is not allowed.",
   367→          );
   368→        }
   369→        candidateIds = this.applyFilter(state, candidateIds, f);
   370→      }
   371→    }
   372→
   373→    // --- Text query: tokenize + score (BM25-ish) ---
   374→    const queryTokens = query.query ? tokenize(query.query) : [];
   375→    const phraseTokens = query.phrase ? tokenize(query.phrase) : [];
   376→
   377→    const avgdl = state.docs.size > 0 ? state.totalTokens / state.docs.size : 0;
   378→    const scored: { id: string; score: number; doc: SearchDocument }[] = [];
   379→
   380→    for (const id of candidateIds) {
   381→      const idx = state.docs.get(id)!;
   382→      let score = 0;
   383→      let matched = false;
   384→
   385→      if (queryTokens.length > 0) {
   386→        for (const qt of queryTokens) {
   387→          const postings = state.inverted.get(qt);
   388→          if (!postings || !postings.has(id)) continue;
   389→          matched = true;
   390→          const tf = idx.tf.get(qt) ?? 0;
   391→          const df = postings.size;
   392→          const N = state.docs.size;
   393→          const idf = Math.log(1 + (N - df + 0.5) / (df + 0.5));
   394→          let docLen = 0;
   395→          for (const toks of idx.fieldTokens.values()) docLen += toks.length;
   396→          const denom = tf + K1 * (1 - B + B * (docLen / (avgdl || 1)));
   397→          score += (idf * (tf * (K1 + 1))) / (denom || 1);
   398→        }
   399→        // OR semantics: a doc must match at least one query token to qualify.
   400→        if (!matched) continue;
   401→      } else if (query.phrase) {
   402→        // Phrase-only query: include docs that contain the phrase.
   403→        if (!this.containsPhrase(idx, phraseTokens)) continue;
   404→        matched = true;
   405→        score = 1;
   406→      }
   407→      // else: no text query — match-all (filtered listing). Score 0.
   408→
   409→      // Phrase filter on top of token query.
   410→      if (query.phrase && !this.containsPhrase(idx, phraseTokens)) {
   411→        continue;
   412→      }
   413→
   414→      scored.push({ id, score, doc: idx.doc });
   415→    }
   416→
   417→    // --- Sort ---
   418→    if (query.sort) {
   419→      if (def.sortableFields.length && !def.sortableFields.includes(query.sort.field)) {
   420→        throw new ValidationError(
   421→          "eks.error.search.field_not_sortable",
   422→          `Field '${query.sort.field}' is not sortable in index '${indexName}'`,
   423→          "Search sort is not allowed.",
   424→        );
   425→      }
   426→      const dir = query.sort.direction === "asc" ? 1 : -1;
   427→      const field = query.sort.field;
   428→      scored.sort((a, b) => {
   429→        const av = a.doc.fields[field];
   430→        const bv = b.doc.fields[field];
   431→        if (av === bv) return 0;
   432→        if (av === null || av === undefined) return 1;
   433→        if (bv === null || bv === undefined) return -1;
   434→        if (typeof av === "number" && typeof bv === "number") {
   435→          return (av - bv) * dir;
   436→        }
   437→        return String(av).localeCompare(String(bv)) * dir;
   438→      });
   439→    } else if (queryTokens.length > 0 || query.phrase) {
   440→      // Default: relevance desc.
   441→      scored.sort((a, b) => b.score - a.score);
   442→    }
   443→
   444→    const total = scored.length;
   445→    const page = scored.slice(offset, offset + limit);
   446→
   447→    // --- Highlights ---
   448→    const highlightFields =
   449→      query.highlights ?? (def.searchableFields.length ? def.searchableFields.slice(0, 2) : []);
   450→
   451→    const hits: SearchHit[] = page.map(({ id, score, doc }) => {
   452→      const idx = state.docs.get(id)!;
   453→      const highlights: SearchHighlight[] = [];
   454→      if (queryTokens.length > 0 || phraseTokens.length > 0) {
   455→        const needles = new Set([...queryTokens, ...phraseTokens]);
   456→        for (const field of highlightFields) {
   457→          const toks = idx.fieldTokens.get(field) ?? [];
   458→          if (toks.length === 0) continue;
   459→          const snippet = buildSnippet(toks, needles);
   460→          if (snippet) highlights.push({ field, snippet });
   461→        }
   462→      }
   463→      return { id, score, doc, highlights: highlights.length ? highlights : undefined };
   464→    });
   465→
   466→    // --- Aggregations (terms only for the default adapter) ---
   467→    let aggregations: Record<string, Record<string, number>> | undefined;
   468→    if (query.aggregations && query.aggregations.length > 0) {
   469→      aggregations = {};
   470→      for (const agg of query.aggregations) {
   471→        if (def.filterableFields.length && !def.filterableFields.includes(agg.field)) {
   472→          throw new ValidationError(
   473→            "eks.error.search.field_not_aggregatable",
   474→            `Field '${agg.field}' is not aggregatable in index '${indexName}'`,
   475→            "Aggregation is not allowed.",
   476→          );
   477→        }
   478→        const buckets: Record<string, number> = {};
   479→        if (agg.kind === "terms") {
   480→          for (const { doc } of scored) {
   481→            const v = doc.fields[agg.field];
   482→            if (v === null || v === undefined) continue;
   483→            if (Array.isArray(v)) {
   484→              for (const item of v) {
   485→                const key = String(item);
   486→                buckets[key] = (buckets[key] ?? 0) + 1;
   487→              }
   488→            } else {
   489→              const key = String(v);
   490→              buckets[key] = (buckets[key] ?? 0) + 1;
   491→            }
   492→          }
   493→        } else if (agg.kind === "range" && agg.ranges) {
   494→          for (const r of agg.ranges) buckets[r.key] = 0;
   495→          for (const { doc } of scored) {
   496→            const v = doc.fields[agg.field];
   497→            if (typeof v !== "number") continue;
   498→            for (const r of agg.ranges) {
   499→              const from = r.from ?? -Infinity;
   500→              const to = r.to ?? Infinity;
   501→              if (v >= from && v < to) {
   502→                buckets[r.key] = (buckets[r.key] ?? 0) + 1;
   503→                break;
   504→              }
   505→            }
   506→          }
   507→        } else if (agg.kind === "stats") {
   508→          let count = 0;
   509→          let sum = 0;
   510→          let min = Infinity;
   511→          let max = -Infinity;
   512→          for (const { doc } of scored) {
   513→            const v = doc.fields[agg.field];
   514→            if (typeof v !== "number") continue;
   515→            count++;
   516→            sum += v;
   517→            if (v < min) min = v;
   518→            if (v > max) max = v;
   519→          }
   520→          buckets.count = count;
   521→          buckets.sum = sum;
   522→          buckets.min = count ? min : 0;
   523→          buckets.max = count ? max : 0;
   524→          buckets.avg = count ? sum / count : 0;
   525→        }
   526→        aggregations[agg.name] = buckets;
   527→      }
   528→    }
   529→
   530→    const tookMs = getClock().epochMs() - started;
   531→    return {
   532→      indexName,
   533→      hits,
   534→      total,
   535→      tookMs,
   536→      aggregations,
   537→    };
   538→  }
   539→
   540→  private applyFilter(state: IndexState, ids: Set<string>, f: SearchFilter): Set<string> {
   541→    const out = new Set<string>();
   542→    for (const id of ids) {
   543→      const doc = state.docs.get(id)!.doc;
   544→      const v = doc.fields[f.field];
   545→      switch (f.op) {
   546→        case "exists":
   547→          if (v !== null && v !== undefined) out.add(id);
   548→          break;
   549→        case "eq":
   550→          if (Array.isArray(v)) {
   551→            if (v.some((x) => String(x) === String(f.value))) out.add(id);
   552→          } else if (String(v) === String(f.value)) {
   553→            out.add(id);
   554→          }
   555→          break;
   556→        case "ne":
   557→          if (Array.isArray(v)) {
   558→            if (!v.some((x) => String(x) === String(f.value))) out.add(id);
   559→          } else if (String(v) !== String(f.value)) {
   560→            out.add(id);
   561→          }
   562→          break;
   563→        case "in": {
   564→          const vals = (f.value as readonly SearchFieldValue[] | undefined) ?? [];
   565→          const set = new Set(vals.map((x) => String(x)));
   566→          if (Array.isArray(v)) {
   567→            if (v.some((x) => set.has(String(x)))) out.add(id);
   568→          } else if (set.has(String(v))) {
   569→            out.add(id);
   570→          }
   571→          break;
   572→        }
   573→        case "gt":
   574→        case "gte":
   575→        case "lt":
   576→        case "lte": {
   577→          const target = f.value as number | string | undefined;
   578→          if (target === undefined) break;
   579→          if (typeof v === "number" && typeof target === "number") {
   580→            if (f.op === "gt" && v > target) out.add(id);
   581→            else if (f.op === "gte" && v >= target) out.add(id);
   582→            else if (f.op === "lt" && v < target) out.add(id);
   583→            else if (f.op === "lte" && v <= target) out.add(id);
   584→          } else if (typeof v === "string" && typeof target === "string") {
   585→            const cmp = v.localeCompare(target);
   586→            if (f.op === "gt" && cmp > 0) out.add(id);
   587→            else if (f.op === "gte" && cmp >= 0) out.add(id);
   588→            else if (f.op === "lt" && cmp < 0) out.add(id);
   589→            else if (f.op === "lte" && cmp <= 0) out.add(id);
   590→          }
   591→          break;
   592→        }
   593→        case "prefix":
   594→          if (typeof v === "string" && typeof f.value === "string") {
   595→            if (v.toLowerCase().startsWith(f.value.toLowerCase())) out.add(id);
   596→          }
   597→          break;
   598→      }
   599→    }
   600→    return out;
   601→  }
   602→
   603→  /** True if the phrase's tokens appear consecutively in any searchable field. */
   604→  private containsPhrase(idx: IndexedDoc, phraseTokens: string[]): boolean {
   605→    if (phraseTokens.length === 0) return true;
   606→    for (const toks of idx.fieldTokens.values()) {
   607→      for (let i = 0; i + phraseTokens.length <= toks.length; i++) {
   608→        let match = true;
   609→        for (let j = 0; j < phraseTokens.length; j++) {
   610→          if (toks[i + j] !== phraseTokens[j]) {
   611→            match = false;
   612→            break;
   613→          }
   614→        }
   615→        if (match) return true;
   616→      }
   617→    }
   618→    return false;
   619→  }
   620→
   621→  /** Test/maintenance hook. */
   622→  clear(indexName?: string): void {
   623→    if (indexName) {
   624→      this.indices.delete(indexName);
   625→    } else {
   626→      this.indices.clear();
   627→    }
   628→  }
   629→
   630→  docCount(indexName: string): number {
   631→    return this.indices.get(indexName)?.docs.size ?? 0;
   632→  }
   633→}
   634→
   635→function buildSnippet(toks: string[], needles: Set<string>, radius = 6): string {
   636→  if (toks.length === 0) return "";
   637→  let firstHit = -1;
   638→  for (let i = 0; i < toks.length; i++) {
   639→    if (needles.has(toks[i])) {
   640→      firstHit = i;
   641→      break;
   642→    }
   643→  }
   644→  if (firstHit === -1) return "";
   645→  const start = Math.max(0, firstHit - radius);
   646→  const end = Math.min(toks.length, firstHit + radius + 1);
   647→  const slice = toks.slice(start, end);
   648→  const marked = slice.map((t) => (needles.has(t) ? `<mark>${t}</mark>` : t));
   649→  let snippet = marked.join(" ");
   650→  if (start > 0) snippet = `… ${snippet}`;
   651→  if (end < toks.length) snippet = `${snippet} …`;
   652→  return snippet;
   653→}

   659→export interface SearchManagerStats {
   660→  readonly registeredProviders: number;
   661→  readonly registeredIndices: number;
   662→  readonly defaultProvider: string | null;
   663→}
   664→
   665→export class SearchManager {
   666→  private readonly providers = new Map<string, SearchProvider>();
   667→  private readonly indices = new Map<string, SearchIndex>();
   668→  private defaultProvider: string | null = null;
   669→
   670→  constructor() {
   671→    const mem = new InMemorySearchProvider();
   672→    this.providers.set(mem.name, mem);
   673→    this.defaultProvider = mem.name;
   674→    // Register the well-known catalog.
   675→    for (const name of Object.keys(INDEX_DEFINITIONS) as SearchIndexName[]) {
   676→      this.indices.set(INDEX_DEFINITIONS[name].name, INDEX_DEFINITIONS[name]);
   677→    }
   678→  }
   679→
   680→  registerProvider(name: string, provider: SearchProvider): void {
   681→    this.providers.set(name, provider);
   682→    if (this.defaultProvider === null) this.defaultProvider = name;
   683→  }
   684→
   685→  setDefault(name: string): void {
   686→    if (!this.providers.has(name)) {
   687→      throw new NotFoundError(
   688→        "eks.error.search.provider_not_found",
   689→        `Search provider '${name}' is not registered`,
   690→        "Search provider is not configured.",
   691→      );
   692→    }
   693→    this.defaultProvider = name;
   694→  }
   695→
   696→  getDefault(): SearchProvider {
   697→    if (!this.defaultProvider) {
   698→      throw new NotFoundError(
   699→        "eks.error.search.no_default_provider",
   700→        "No default search provider is configured",
   701→        "Search is not configured.",
   702→      );
   703→    }
   704→    const p = this.providers.get(this.defaultProvider);
   705→    if (!p) {
   706→      throw new NotFoundError(
   707→        "eks.error.search.provider_not_found",
   708→        `Default search provider '${this.defaultProvider}' is missing`,
   709→        "Search is not configured.",
   710→      );
   711→    }
   712→    return p;
   713→  }
   714→
   715→  getProvider(name: string): SearchProvider | undefined {
   716→    return this.providers.get(name);
   717→  }
   718→
   719→  listProviders(): readonly string[] {
   720→    return [...this.providers.keys()];
   721→  }
   722→
   723→  registerIndex(index: SearchIndex): void {
   724→    this.indices.set(index.name, index);
   725→  }
   726→
   727→  listIndices(): readonly SearchIndex[] {
   728→    return [...this.indices.values()];
   729→  }
   730→
   731→  getIndex(name: string): SearchIndex | undefined {
   732→    return this.indices.get(name);
   733→  }
   734→
   735→  /** Validate a doc against an index's declared fields. */
   736→  private validateDoc(indexName: string, doc: SearchDocument): SearchIndex {
   737→    const idx = this.indices.get(indexName);
   738→    if (!idx) {
   739→      throw new NotFoundError(
   740→        "eks.error.search.index_not_found",
   741→        `Search index '${indexName}' is not registered`,
   742→        "Search index is not configured.",
   743→      );
   744→    }
   745→    const declared = new Set([
   746→      ...idx.searchableFields,
   747→      ...idx.filterableFields,
   748→      ...idx.sortableFields,
   749→    ]);
   750→    if (declared.size > 0) {
   751→      for (const key of Object.keys(doc.fields)) {
   752→        if (!declared.has(key)) {
   753→          // Allow extra fields; they're just not searchable/filterable/sortable.
   754→          // Real engines index arbitrary fields; we keep them as-is.
   755→        }
   756→      }
   757→    }
   758→    return idx;
   759→  }
   760→
   761→  async index(indexName: string, doc: SearchDocument): Promise<void> {
   762→    this.validateDoc(indexName, doc);
   763→    await this.getDefault().index(indexName, doc);
   764→  }
   765→
   766→  async search(indexName: string, query: SearchQuery): Promise<SearchResult> {
   767→    if (!this.indices.has(indexName)) {
   768→      throw new NotFoundError(
   769→        "eks.error.search.index_not_found",
   770→        `Search index '${indexName}' is not registered`,
   771→        "Search index is not configured.",
   772→      );
   773→    }
   774→    return this.getDefault().search(indexName, query);
   775→  }
   776→
   777→  async delete(indexName: string, id: string): Promise<boolean> {
   778→    if (!this.indices.has(indexName)) {
   779→      throw new NotFoundError(
   780→        "eks.error.search.index_not_found",
   781→        `Search index '${indexName}' is not registered`,
   782→        "Search index is not configured.",
   783→      );
   784→    }
   785→    return this.getDefault().delete(indexName, id);
   786→  }
   787→
   788→  async reindex(indexName: string): Promise<number> {
   789→    if (!this.indices.has(indexName)) {
   790→      throw new NotFoundError(
   791→        "eks.error.search.index_not_found",
   792→        `Search index '${indexName}' is not registered`,
   793→        "Search index is not configured.",
   794→      );
   795→    }
   796→    return this.getDefault().reindex(indexName);
   797→  }
   798→
   799→  stats(): SearchManagerStats {
   800→    return {
   801→      registeredProviders: this.providers.size,
   802→      registeredIndices: this.indices.size,
   803→      defaultProvider: this.defaultProvider,
   804→    };
   805→  }
   806→}
