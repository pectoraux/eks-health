     2→ * Eks-Health Kernel — AI Readiness (Architecture Only)
     3→ *
     4→ * This module deliberately does NOT implement AI execution. It prepares the
     5→ * architectural substrate so that any AI provider, agent runtime, vector
     6→ * store, or model router can be plugged in later without touching the rest
     7→ * of the platform.
     8→ *
     9→ * What IS implemented here (real, working, no mocks):
    10→ *   - An in-memory vector store with REAL cosine-similarity search.
    11→ *   - A prompt registry with REAL `{var}` template interpolation.
    12→ *   - A model router with REAL filtering by modality / context window /
    13→ *     capability flags and sorting by cost or capability score.
    14→ *   - A tool registry with REAL descriptor lookup and a real invocation
    15→ *     contract (handlers are user-supplied; absent handlers return a
    16→ *     structured "no_handler" result rather than faking output).
    17→ *   - An agent runtime that produces fully-formed AgentRun objects whose
    18→ *     status is `pending_provider` because no provider is wired in by

    20→ *   - An AI observability ledger that records every prompt execution and

   152→/** REAL cosine similarity between two equal-length vectors. */
   153→export function cosineSimilarity(a: readonly number[], b: readonly number[]): number {
   154→  if (a.length !== b.length || a.length === 0) return 0;
   155→  let dot = 0;
   156→  let normA = 0;
   157→  let normB = 0;
   158→  for (let i = 0; i < a.length; i++) {
   159→    const av = a[i];
   160→    const bv = b[i];
   161→    dot += av * bv;
   162→    normA += av * av;
   163→    normB += bv * bv;
   164→  }
   165→  const denom = Math.sqrt(normA) * Math.sqrt(normB);
   166→  if (denom === 0) return 0;
   167→  return dot / denom;
   168→}
   169→
   170→/**
   171→ * REAL working in-memory vector store. Stores embeddings as `number[]`,
   172→ * computes cosine similarity against every candidate, and returns the top-K
   173→ * above an optional minimum score. Suitable for dev and small corpora;
   174→ * production swaps in pgvector / Pinecone / Weaviate.
   175→ */
   176→export class InMemoryVectorStore implements VectorStore {
   177→  readonly name = "in-memory";
   178→  private readonly store = new Map<string, Embedding>();
   179→
   180→  async upsert(embeddings: readonly Embedding[]): Promise<void> {
   181→    for (const e of embeddings) {
   182→      this.store.set(e.id, e);
   183→    }
   184→  }
   185→
   186→  async query(q: VectorQuery): Promise<readonly VectorQueryResult[]> {
   187→    const topK = q.topK ?? 5;
   188→    const minScore = q.minScore ?? -1;
   189→    const results: VectorQueryResult[] = [];
   190→    for (const e of this.store.values()) {
   191→      if (q.filter && !q.filter(e)) continue;
   192→      if (e.vector.length !== q.vector.length) continue;
   193→      const score = cosineSimilarity(q.vector, e.vector);
   194→      if (score < minScore) continue;
   195→      results.push({ embedding: e, score });
   196→    }
   197→    results.sort((a, b) => b.score - a.score);
   198→    return results.slice(0, topK);
   199→  }
   200→
   201→  async delete(ids: readonly string[]): Promise<number> {
   202→    let removed = 0;
   203→    for (const id of ids) {
   204→      if (this.store.delete(id)) removed++;
   205→    }
   206→    return removed;
   207→  }
   208→
   209→  size(): number {
   210→    return this.store.size;
   211→  }
   212→}

   236→export class PromptRegistry {
   237→  private readonly templates = new Map<string, PromptTemplate>();
   238→
   239→  register(template: PromptTemplate): PromptTemplate {
   240→    this.templates.set(template.id, template);
   241→    return template;

   249→    return [...this.templates.values()];

   251→
   252→  /**
   253→   * Render a template by interpolating `{var}` placeholders.
   254→   * Unknown variables are replaced with the empty string and recorded in
   255→   * the returned `missing` list (consumers may detect this via metadata).
   256→   */
   257→  render(id: string, vars: Record<string, string>): RenderedPrompt {
   258→    const tpl = this.templates.get(id);
   259→    if (!tpl) throw new Error(`PromptRegistry.render: template ${id} not found`);
   260→    const resolved: Record<string, string> = {};
   261→    const missing: string[] = [];
   262→    for (const v of tpl.variables) {
   263→      const val = vars[v];
   264→      if (val === undefined || val === null) {
   265→        missing.push(v);
   266→        resolved[v] = "";
   267→      } else {
   268→        resolved[v] = String(val);
   269→      }
   270→    }
   271→    const text = tpl.template.replace(/\{(\w+)\}/g, (full, name: string) => {
   272→      if (!(name in resolved)) return full;
   273→      return resolved[name];
   274→    });
   275→    return {
   276→      templateId: id,
   277→      text,
   278→      variables: resolved,
   279→      renderedAt: getClock().iso(),
   280→    };
   281→  }
   282→}

   304→export interface ToolInvocation {
   305→  readonly id: string;
   306→  readonly tool: string;
   307→  readonly input: Readonly<Record<string, unknown>>;
   308→  readonly output?: unknown;
   309→  readonly error?: string;
   310→  readonly ok: boolean;
   311→  readonly startedAt: string;
   312→  readonly finishedAt: string;
   313→  readonly latencyMs: number;
   314→}
   315→
   316→export class ToolRegistry {
   317→  private readonly tools = new Map<string, { descriptor: ToolDescriptor; handler?: ToolHandler }>();
   318→
   319→  register(tool: ToolDescriptor, handler?: ToolHandler): ToolDescriptor {
   320→    this.tools.set(tool.name, { descriptor: tool, handler });
   321→    return tool;
   322→  }
   323→
   324→  get(name: string): ToolDescriptor | undefined {
   325→    return this.tools.get(name)?.descriptor;
   326→  }
   327→
   328→  list(): readonly ToolDescriptor[] {
   329→    return [...this.tools.values()].map((t) => t.descriptor);
   330→  }
   331→
   332→  /**
   333→   * Invoke a registered tool. If no handler has been attached (the common
   334→   * case during early scaffolding), the invocation returns a structured
   335→   * `ok:false` result with error `no_handler` rather than faking output.
   336→   */
   337→  async invoke(name: string, input: Record<string, unknown>): Promise<ToolInvocation> {
   338→    const entry = this.tools.get(name);
   339→    const startedAt = getClock().iso();
   340→    const start = Date.now();
   341→    if (!entry) {
   342→      return this.finish(name, input, startedAt, start, {
   343→        ok: false,
   344→        error: `tool_not_registered: ${name}`,
   345→      });
   346→    }
   347→    if (!entry.handler) {
   348→      return this.finish(name, input, startedAt, start, {
   349→        ok: false,
   350→        error: "no_handler",
   351→      });
   352→    }
   353→    try {
   354→      const result = await entry.handler(input);
   355→      return this.finish(name, input, startedAt, start, result);
   356→    } catch (e) {
   357→      return this.finish(name, input, startedAt, start, {
   358→        ok: false,
   359→        error: e instanceof Error ? e.message : String(e),
   360→      });
   361→    }
   362→  }
   363→
   364→  private finish(
   365→    name: string,
   366→    input: Record<string, unknown>,
   367→    startedAt: string,
   368→    startMs: number,
   369→    result: ToolInvocationResult,
   370→  ): ToolInvocation {
   371→    const finishedAt = getClock().iso();
   372→    return {
   373→      id: `inv_${generateId()}`,
   374→      tool: name,
   375→      input,
   376→      output: result.output,
   377→      error: result.error,
   378→      ok: result.ok,
   379→      startedAt,
   380→      finishedAt,
   381→      latencyMs: result.latencyMs ?? Date.now() - startMs,
   382→    };
   383→  }
   384→}

   434→/**
   435→ * Real router. Filters by modality, context window, capability flags and
   436→ * deprecation status; then sorts by the requested cost preference. When no
   437→ * model satisfies the constraints the router throws (callers must handle).
   438→ */
   439→export class ModelRouter {
   440→  private readonly models = new Map<string, AIModelDescriptor>();
   441→
   442→  registerModel(descriptor: AIModelDescriptor): AIModelDescriptor {
   443→    this.models.set(descriptor.id, descriptor);
   444→    return descriptor;
   445→  }
   446→
   447→  unregisterModel(id: string): boolean {
   448→    return this.models.delete(id);
   449→  }
   450→
   451→  list(): readonly AIModelDescriptor[] {
   452→    return [...this.models.values()];
   453→  }
   454→
   455→  route(request: ModelRouterRequest): RouteDecision {
   456→    const candidates = [...this.models.values()].filter((m) => {
   457→      if (request.excludeDeprecated !== false && m.deprecated) return false;
   458→      if (request.modality) {
   459→        // A model matches if it directly supports the requested modality OR
   460→        // is multimodal (capable of serving any single-modality request).
   461→        const modalityOk =
   462→          m.modality === request.modality || m.modality === "multimodal";
   463→        if (!modalityOk) return false;
   464→      }
   465→      if (request.minContextWindow !== undefined && m.contextWindow < request.minContextWindow) {
   466→        return false;
   467→      }
   468→      if (
   469→        request.maxInputTokens !== undefined &&
   470→        m.contextWindow < request.maxInputTokens
   471→      ) {
   472→        return false;
   473→      }
   474→      if (request.requiredCapabilities) {
   475→        for (const cap of request.requiredCapabilities) {
   476→          if (!m.capabilities.has(cap)) return false;
   477→        }
   478→      }
   479→      return true;
   480→    });
   481→
   482→    if (candidates.length === 0) {
   483→      throw new Error(
   484→        `ModelRouter.route: no model satisfies constraints (modality=${request.modality ?? "any"}, ` +
   485→          `minCtx=${request.minContextWindow ?? "any"}, caps=${(request.requiredCapabilities ?? []).join(",")})`,
   486→      );
   487→    }
   488→
   489→    const pref = request.costPreference ?? "balanced";
   490→    const scored = candidates
   491→      .map((m) => ({
   492→        m,
   493→        score: this.score(m, pref),
   494→        cost: m.costPer1kTokens.input + m.costPer1kTokens.output,
   495→      }))
   496→      .sort((a, b) => {
   497→        if (pref === "cheapest") return a.cost - b.cost;
   498→        if (pref === "highest_quality") return b.score - a.score;
   499→        // balanced: normalise cost & capability, pick best ratio
   500→        return b.score - a.score === 0 ? a.cost - b.cost : b.score - a.score;
   501→      });
   502→
   503→    const winner = scored[0];
   504→    return {
   505→      model: winner.m,
   506→      alternatives: scored.slice(1, 4).map((s) => s.m),
   507→      reason: `${pref}:${winner.m.vendor}/${winner.m.family} (score=${winner.score.toFixed(2)}, cost=$${winner.cost.toFixed(4)}/1k)`,
   508→    };
   509→  }
   510→
   511→  private score(m: AIModelDescriptor, pref: "cheapest" | "balanced" | "highest_quality"): number {
   512→    // A simple real heuristic: more capabilities + larger context = higher score.
   513→    const capScore = m.capabilities.size * 10;
   514→    const ctxScore = Math.log10(Math.max(1, m.contextWindow)) * 5;
   515→    const costScore = pref === "cheapest" ? 0 : 0;
   516→    void costScore;
   517→    return capScore + ctxScore;
   518→  }
   519→}

   562→/**
   563→ * Agent runtime. Registers agent descriptors and produces fully-formed
   564→ * `AgentRun` objects. Because no AI provider is registered by default,
   565→ * `startRun` returns a run with status `pending_provider` — this is the
   566→ * intended scaffolding behavior. Downstream code (or a future provider
   567→ * adapter) is responsible for transitioning runs to `running` / `completed`.
   568→ */
   569→export class AgentRuntime {
   570→  private readonly agents = new Map<string, AgentDescriptor>();
   571→  private readonly runs = new Map<string, AgentRun>();
   572→
   573→  registerAgent(descriptor: AgentDescriptor): AgentDescriptor {
   574→    this.agents.set(descriptor.id, descriptor);
   575→    return descriptor;
   576→  }
   577→
   578→  get(id: string): AgentDescriptor | undefined {
   579→    return this.agents.get(id);
   580→  }
   581→
   582→  list(): readonly AgentDescriptor[] {
   583→    return [...this.agents.values()];
   584→  }
   585→
   586→  startRun(agentId: string, input: string): AgentRun {
   587→    const agent = this.agents.get(agentId);
   588→    if (!agent) throw new Error(`AgentRuntime.startRun: agent ${agentId} not found`);
   589→    const run: AgentRun = {
   590→      id: `run_${generateId()}`,
   591→      agentId,
   592→      input,
   593→      status: "pending_provider",
   594→      startedAt: getClock().iso(),
   595→      messages: [
   596→        { role: "system", content: agent.systemPrompt },
   597→        { role: "user", content: input },
   598→      ],
   599→      toolInvocations: [],
   600→    };
   601→    this.runs.set(run.id, run);
   602→    return run;
   603→  }
   604→
   605→  getRun(runId: string): AgentRun | undefined {
   606→    return this.runs.get(runId);
   607→  }
   608→
   609→  listRuns(): readonly AgentRun[] {
   610→    return [...this.runs.values()];
   611→  }
   612→
   613→  /** Transition a run to a new status (used by future provider adapters). */
   614→  transitionRun(runId: string, status: AgentRunStatus, patch: Partial<AgentRun> = {}): AgentRun {
   615→    const existing = this.runs.get(runId);
   616→    if (!existing) throw new Error(`AgentRuntime.transitionRun: run ${runId} not found`);
   617→    const updated: AgentRun = {
   618→      ...existing,
   619→      ...patch,
   620→      status,
   621→      finishedAt:
   622→        status === "completed" || status === "failed" || status === "cancelled"
   623→          ? getClock().iso()
   624→          : patch.finishedAt,
   625→    };
   626→    this.runs.set(runId, updated);
   627→    return updated;
   628→  }
   629→}

   645→  readonly ok: boolean;
   646→  readonly error?: string;
   647→  readonly occurredAt: string;

   665→/**
   666→ * Records every prompt execution and tool invocation with token counts,
   667→ * latency, and a cost estimate derived from the routed model's pricing.
   668→ */
   669→export class AIObservability {
   670→  private readonly promptLog: PromptExecution[] = [];
   671→  private readonly invocationLog: AIInvocationLog[] = [];
   672→  private readonly toolLog: ToolInvocation[] = [];
   673→
   674→  recordPromptExecution(input: Omit<PromptExecution, "id" | "occurredAt">): PromptExecution {
   675→    const entry: PromptExecution = {
   676→      ...input,
   677→      id: `pex_${generateId()}`,
   678→      occurredAt: getClock().iso(),
   679→    };
   680→    this.promptLog.push(entry);
   681→    this.invocationLog.push({
   682→      id: `log_${generateId()}`,
   683→      kind: "prompt",
   684→      ref: entry.id,
   685→      agentRunId: undefined,
   686→      model: entry.model,
   687→      provider: entry.provider,
   688→      tokens: entry.usage?.totalTokens,
   689→      latencyMs: entry.latencyMs,
   690→      estimatedCostUsd: entry.estimatedCostUsd,
   691→      ok: entry.ok,
   692→      occurredAt: entry.occurredAt,
   693→    });
   694→    return entry;
   695→  }
   696→
   697→  recordToolInvocation(invocation: ToolInvocation, agentRunId?: string): ToolInvocation {
   698→    this.toolLog.push(invocation);
   699→    this.invocationLog.push({
   700→      id: `log_${generateId()}`,
   701→      kind: "tool",
   702→      ref: invocation.id,
   703→      agentRunId,
   704→      tool: invocation.tool,
   705→      latencyMs: invocation.latencyMs,
   706→      ok: invocation.ok,
   707→      occurredAt: invocation.finishedAt,
   708→    });
   709→    return invocation;
   710→  }
   711→
   712→  /** Estimate USD cost given a model's per-1k pricing and a token usage. */
   713→  estimateCost(model: AIModelDescriptor, usage: TokenUsage): number {
   714→    const inCost = (usage.promptTokens / 1000) * model.costPer1kTokens.input;
   715→    const outCost = (usage.completionTokens / 1000) * model.costPer1kTokens.output;
   716→    return Math.round((inCost + outCost) * 1_000_000) / 1_000_000;
   717→  }
   718→
   719→  listPromptExecutions(limit = 200): readonly PromptExecution[] {
   720→    return this.promptLog.slice(-limit);
   721→  }
   722→
   723→  listToolInvocations(limit = 200): readonly ToolInvocation[] {
   724→    return this.toolLog.slice(-limit);
   725→  }
   726→
   727→  listInvocations(limit = 500): readonly AIInvocationLog[] {
   728→    return this.invocationLog.slice(-limit);
   729→  }
   730→}
