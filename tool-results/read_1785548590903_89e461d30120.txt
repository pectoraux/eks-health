   141→export function TechnicianDashboard({ data, onRefresh }: { data: DashboardData; onRefresh: () => void }) {
   142→  const techs = data.technicians;
   143→  const measurements = data.measurements;
   144→
   145→  // Sub-API state (loaded client-side; dashboard route only ships aggregate stats)
   146→  const [sessions, setSessions] = useState<MeasurementSession[]>([]);
   147→  const [appointments, setAppointments] = useState<Appointment[]>([]);
   148→  const [devices, setDevices] = useState<DeviceRecord[]>([]);
   149→  const [disputes, setDisputes] = useState<DisputeRecord[]>([]);
   150→  const [fraudAlerts, setFraudAlerts] = useState<FraudAlert[]>([]);
   151→  const [loadingSub, setLoadingSub] = useState(true);
   152→  const [subError, setSubError] = useState<string | null>(null);
   153→
   154→  // Record-measurement dialog state
   155→  const [recordOpen, setRecordOpen] = useState(false);
   156→
   157→  const loadSubData = useCallback(async () => {
   158→    setLoadingSub(true);
   159→    setSubError(null);
   160→    try {
   161→      const [sessRes, apptRes, devRes, dispRes, fraudRes] = await Promise.allSettled([
   162→        fetch("/api/technicians/sessions", { cache: "no-store" }),
   163→        fetch("/api/technicians/appointments", { cache: "no-store" }),
   164→        fetch("/api/technicians/devices", { cache: "no-store" }),
   165→        fetch("/api/technicians/disputes", { cache: "no-store" }),
   166→        fetch("/api/technicians/fraud", { cache: "no-store" }),
   167→      ]);
   168→
   169→      // Promise.allSettled wraps the fetch promises; the json() reads happen below.
   170→      const readJson = async <T,>(r: PromiseSettledResult<Response>, fallback: T, label: string): Promise<T> => {
   171→        if (r.status !== "fulfilled") {
   172→          console.warn(`[technician-dashboard] ${label} request rejected`);
   173→          return fallback;
   174→        }
   175→        try {
   176→          const j = (await r.value.json()) as { ok?: boolean; data?: T; error?: { message?: string } };
   177→          if (!j?.ok) {
   178→            console.warn(`[technician-dashboard] ${label} returned ok=false`, j?.error?.message);
   179→            return fallback;
   180→          }
   181→          return (j.data as T) ?? fallback;
   182→        } catch {
   183→          return fallback;
   184→        }
   185→      };
   186→
   187→      const [sess, appt, dev, disp, fraud] = await Promise.all([
   188→        readJson<{ sessions: MeasurementSession[] } | MeasurementSession[]>(sessRes, { sessions: [] }, "sessions"),
   189→        readJson<{ appointments: Appointment[] } | Appointment[]>(apptRes, { appointments: [] }, "appointments"),
   190→        readJson<{ devices: DeviceRecord[] } | DeviceRecord[]>(devRes, { devices: [] }, "devices"),
   191→        readJson<{ disputes: DisputeRecord[] } | DisputeRecord[]>(dispRes, { disputes: [] }, "disputes"),
   192→        readJson<{ alerts: FraudAlert[] } | FraudAlert[]>(fraudRes, { alerts: [] }, "fraud"),
   193→      ]);
   194→
   195→      setSessions(Array.isArray(sess) ? sess : sess.sessions ?? []);
   196→      setAppointments(Array.isArray(appt) ? appt : appt.appointments ?? []);
   197→      setDevices(Array.isArray(dev) ? dev : dev.devices ?? []);
   198→      setDisputes(Array.isArray(disp) ? disp : disp.disputes ?? []);
   199→      setFraudAlerts(Array.isArray(fraud) ? fraud : fraud.alerts ?? []);
   200→    } catch (err) {
   201→      const msg = err instanceof Error ? err.message : "Failed to load technician data";
   202→      setSubError(msg);
   203→      toast({ title: "Load failed", description: msg, variant: "destructive" });
   204→    } finally {
   205→      setLoadingSub(false);
   206→    }
   207→  }, []);
   208→
   209→  useEffect(() => {
   210→    void loadSubData();
   211→  }, [loadSubData]);
   212→
   213→  // Derived stats ---------------------------------------------------------
   214→  const avgRating = useMemo(() => {
   215→    const rated = techs?.list?.filter((t) => typeof t.rating === "number") ?? [];
   216→    if (rated.length === 0) return 0;
   217→    return rated.reduce((s, t) => s + (t.rating ?? 0), 0) / rated.length;
   218→  }, [techs]);
   219→
   220→  const activeSessions = useMemo(() => {
   221→    const activeStates = new Set(["in_progress", "active", "scheduled", "open"]);
   222→    return sessions.filter((s) => activeStates.has(s.status?.toLowerCase())).length;
   223→  }, [sessions]);
   224→
   225→  const handleRecorded = useCallback(() => {
   226→    setRecordOpen(false);
   227→    onRefresh();
   228→    // Sessions list may reference the new measurement; refresh sub-data too.
   229→    void loadSubData();
   230→  }, [onRefresh, loadSubData]);
   231→
   232→  const handleManualRefresh = useCallback(() => {
   233→    onRefresh();
   234→    void loadSubData();
   235→  }, [onRefresh, loadSubData]);
   236→
   237→  return (
   238→    <div className="space-y-6">
   239→      {/* Top stat row + primary action */}
   240→      <div className="flex flex-col gap-3 sm:flex-row sm:items-stretch">
   241→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 flex-1">
   242→          <StatCard
   243→            icon={<Users className="h-4 w-4" />}
   244→            label="Total Technicians"
   245→            value={techs?.stats.total ?? 0}
   246→            hint={`${techs?.stats.active ?? 0} active`}
   247→          />
   248→          <StatCard
   249→            icon={<Stethoscope className="h-4 w-4" />}
   250→            label="Active Sessions"
   251→            value={activeSessions}
   252→            hint={`${sessions.length} total`}
   253→            loading={loadingSub}
   254→          />
   255→          <StatCard
   256→            icon={<Activity className="h-4 w-4" />}
   257→            label="Measurements Recorded"
   258→            value={measurements?.stats.total ?? 0}
   259→            hint="all profiles"
   260→          />
   261→          <StatCard
   262→            icon={<Star className="h-4 w-4" />}
   263→            label="Avg Rating"
   264→            value={avgRating > 0 ? avgRating.toFixed(2) : "—"}
   265→            hint={`${techs?.list?.filter((t) => typeof t.rating === "number").length ?? 0} rated`}
   266→          />
   267→        </div>
   268→        <div className="flex sm:flex-col gap-2 sm:justify-end">
   269→          <Button
   270→            onClick={() => setRecordOpen(true)}
   271→            className="bg-[var(--brand)] text-[var(--brand-foreground)] hover:opacity-90 h-9"
   272→          >
   273→            <Plus className="h-4 w-4 mr-1" /> Record Measurement
   274→          </Button>
   275→          <Button variant="outline" onClick={handleManualRefresh} className="h-9" disabled={loadingSub}>
   276→            <RefreshCw className={`h-4 w-4 mr-1 ${loadingSub ? "animate-spin" : ""}`} />
   277→            Refresh
   278→          </Button>
   279→        </div>
   280→      </div>
   281→
   282→      {subError && (
   283→        <div className="rounded-md border border-destructive/40 bg-destructive/5 p-3 text-xs text-destructive">
   284→          Could not load some technician data: {subError}. Showing partial view.
   285→        </div>
   286→      )}
   287→
   288→      {/* Sessions + Appointments row */}
   289→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   290→        <SessionsCard sessions={sessions} loading={loadingSub} onRefresh={handleManualRefresh} />
   291→        <AppointmentsCard appointments={appointments} loading={loadingSub} />
   292→      </div>
   293→
   294→      {/* Devices + Disputes/Fraud row */}
   295→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   296→        <DevicesCard devices={devices} loading={loadingSub} />
   297→        <DisputesFraudCard disputes={disputes} fraudAlerts={fraudAlerts} loading={loadingSub} />
   298→      </div>
   299→
   300→      {/* Record-measurement dialog */}
   301→      <RecordMeasurementDialog
   302→        open={recordOpen}
   303→        onOpenChange={setRecordOpen}
   304→        onRecorded={handleRecorded}
   305→      />
   306→    </div>
   307→  );
   308→}

   314→function SessionsCard({
   315→  sessions,
   316→  loading,
   317→  onRefresh,
   318→}: {
   319→  sessions: MeasurementSession[];
   320→  loading: boolean;
   321→  onRefresh: () => void;
   322→}) {
   323→  const [expanded, setExpanded] = useState<string | null>(null);
   324→
   325→  const stateVariant = (state: string): "default" | "secondary" | "destructive" | "outline" => {
   326→    const s = state?.toLowerCase();
   327→    if (s === "completed" || s === "verified") return "default";
   328→    if (s === "in_progress" || s === "active" || s === "scheduled" || s === "open") return "secondary";
   329→    if (s === "disputed" || s === "cancelled" || s === "failed") return "destructive";
   330→    return "outline";
   331→  };
   332→
   333→  return (
   334→    <Card>
   335→      <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-3">
   336→        <CardTitle className="text-base flex items-center gap-2">
   337→          <ClipboardList className="h-4 w-4 text-muted-foreground" />
   338→          Measurement Sessions
   339→        </CardTitle>
   340→        <Badge variant="outline" className="text-[10px]">{sessions.length} sessions</Badge>
   341→      </CardHeader>
   342→      <CardContent>
   343→        {loading ? (
   344→          <ListSkeleton rows={4} />
   345→        ) : sessions.length === 0 ? (
   346→          <EmptyState
   347→            icon={<ClipboardList className="h-8 w-8" />}
   348→            message="No measurement sessions recorded yet."
   349→          />
   350→        ) : (
   351→          <div className="space-y-1.5 max-h-96 overflow-y-auto eks-scroll pr-1">
   352→            {sessions.slice(0, 50).map((s) => {
   353→              const isOpen = expanded === s.id;
   354→              return (
   355→                <Collapsible key={s.id} open={isOpen} onOpenChange={(o) => setExpanded(o ? s.id : null)}>
   356→                  <div className="rounded-lg border border-border/60 overflow-hidden">
   357→                    <CollapsibleTrigger asChild>
   358→                      <button
   359→                        type="button"
   360→                        className="w-full flex items-center gap-3 p-3 text-left hover:bg-accent/40 transition-colors"
   361→                      >
   362→                        <Circle
   363→                          className={`h-2.5 w-2.5 shrink-0 ${
   364→                            s.status === "completed" ? "fill-emerald-500 text-emerald-500"
   365→                            : s.status === "disputed" || s.status === "failed" ? "fill-destructive text-destructive"
   366→                            : "fill-amber-500 text-amber-500"
   367→                          }`}
   368→                        />
   369→                        <div className="flex-1 min-w-0">
   370→                          <p className="text-sm font-medium truncate">
   371→                            {labelFor(s.participantId)}
   372→                          </p>
   373→                          <p className="text-[11px] text-muted-foreground truncate">
   374→                            Tech: {labelFor(s.technicianId)} · {labelFor(s.programId)}
   375→                          </p>
   376→                        </div>
   377→                        <Badge variant={stateVariant(s.status)} className="text-[10px] capitalize">
   378→                          {s.status}
   379→                        </Badge>
   380→                        {isOpen ? (
   381→                          <ChevronDown className="h-4 w-4 text-muted-foreground shrink-0" />
   382→                        ) : (
   383→                          <ChevronRight className="h-4 w-4 text-muted-foreground shrink-0" />
   384→                        )}
   385→                      </button>
   386→                    </CollapsibleTrigger>
   387→                    <CollapsibleContent>
   388→                      <div className="border-t border-border/60 bg-muted/30 p-3 grid grid-cols-2 gap-3 text-xs">
   389→                        <Detail label="Session ID" value={s.id} mono />
   390→                        <Detail label="Participant" value={s.participantId} mono />
   391→                        <Detail label="Technician" value={s.technicianId} mono />
   392→                        <Detail label="Program" value={s.programId} mono />
   393→                        <Detail
   394→                          label="Scheduled"
   395→                          value={fmtDate(s.scheduledAt)}
   396→                        />
   397→                        <Detail
   398→                          label="Completed"
   399→                          value={s.completedAt ? fmtDate(s.completedAt) : "—"}
   400→                        />
   401→                        <Detail
   402→                          label="Measurements"
   403→                          value={String(s.measurementCount)}
   404→                          icon={<Activity className="h-3 w-3" />}
   405→                        />
   406→                        <Detail
   407→                          label="Evidence"
   408→                          value={String(s.evidenceCount)}
   409→                          icon={<ShieldCheck className="h-3 w-3" />}
   410→                        />
   411→                      </div>
   412→                    </CollapsibleContent>
   413→                  </div>
   414→                </Collapsible>
   415→              );
   416→            })}
   417→          </div>
   418→        )}
   419→        {sessions.length > 0 && (
   420→          <div className="mt-3 flex justify-end">
   421→            <Button size="sm" variant="ghost" className="h-7 text-xs" onClick={onRefresh}>
   422→              <RefreshCw className="h-3 w-3 mr-1" /> Reload
   423→            </Button>
   424→          </div>
   425→        )}
   426→      </CardContent>
   427→    </Card>
   428→  );
   429→}

   435→function AppointmentsCard({
   436→  appointments,
   437→  loading,
   438→}: {
   439→  appointments: Appointment[];
   440→  loading: boolean;
   441→}) {
   442→  const [scheduleOpen, setScheduleOpen] = useState(false);
   443→
   444→  const statusVariant = (state: string): "default" | "secondary" | "destructive" | "outline" => {
   445→    const s = state?.toLowerCase();
   446→    if (s === "confirmed" || s === "scheduled" || s === "completed") return "default";
   447→    if (s === "pending" || s === "in_progress") return "secondary";
   448→    if (s === "cancelled" || s === "no_show") return "destructive";
   449→    return "outline";
   450→  };
   451→
   452→  const sorted = useMemo(() => {
   453→    return [...appointments].sort((a, b) => {
   454→      const ta = new Date(a.scheduledAt).getTime();
   455→      const tb = new Date(b.scheduledAt).getTime();
   456→      if (Number.isNaN(ta) && Number.isNaN(tb)) return 0;
   457→      if (Number.isNaN(ta)) return 1;
   458→      if (Number.isNaN(tb)) return -1;
   459→      return ta - tb;
   460→    });
   461→  }, [appointments]);
   462→
   463→  return (
   464→    <Card>
   465→      <CardHeader className="flex flex-row items-center justify-between space-y-0 pb-3">
   466→        <CardTitle className="text-base flex items-center gap-2">
   467→          <Calendar className="h-4 w-4 text-muted-foreground" />
   468→          Appointments
   469→        </CardTitle>
   470→        <div className="flex items-center gap-2">
   471→          <Badge variant="outline" className="text-[10px]">{appointments.length} total</Badge>
   472→          <Button
   473→            size="sm"
   474→            variant="outline"
   475→            className="h-7 text-xs"
   476→            onClick={() => setScheduleOpen(true)}
   477→          >
   478→            <Plus className="h-3 w-3 mr-1" /> Schedule New
   479→          </Button>
   480→        </div>
   481→      </CardHeader>
   482→      <CardContent>
   483→        {loading ? (
   484→          <ListSkeleton rows={4} />
   485→        ) : sorted.length === 0 ? (
   486→          <EmptyState
   487→            icon={<Calendar className="h-8 w-8" />}
   488→            message="No appointments on the calendar."
   489→          />
   490→        ) : (
   491→          <div className="space-y-1.5 max-h-96 overflow-y-auto eks-scroll pr-1">
   492→            {sorted.slice(0, 50).map((a) => (
   493→              <div
   494→                key={a.id}
   495→                className="flex items-center gap-3 rounded-lg border border-border/60 p-3"
   496→              >
   497→                <CalendarClock className="h-4 w-4 shrink-0 text-muted-foreground" />
   498→                <div className="flex-1 min-w-0">
   499→                  <p className="text-sm font-medium truncate">
   500→                    {labelFor(a.participantId)}
   501→                  </p>
   502→                  <p className="text-[11px] text-muted-foreground truncate">
   503→                    {fmtDate(a.scheduledAt)}
   504→                    {a.durationMinutes ? ` · ${a.durationMinutes}m` : ""}
   505→                    {a.sessionType ? ` · ${a.sessionType}` : ""}
   506→                  </p>
   507→                </div>
   508→                <Badge variant={statusVariant(a.status)} className="text-[10px] capitalize">
   509→                  {a.status}
   510→                </Badge>
   511→              </div>
   512→            ))}
   513→          </div>
   514→        )}
   515→      </CardContent>
   516→
   517→      <Dialog open={scheduleOpen} onOpenChange={setScheduleOpen}>
   518→        <DialogContent>
   519→          <DialogHeader>
   520→            <DialogTitle>Schedule New Appointment</DialogTitle>
   521→            <DialogDescription>
   522→              Capture basic appointment details. Full scheduling will be wired to the
   523→              technician appointments API once write endpoints ship.
   524→            </DialogDescription>
   525→          </DialogHeader>
   526→          <ScheduleFormStub
   527→            onSubmit={() => {
   528→              toast({
   529→                title: "Saved locally",
   530→                description:
   531→                  "Appointment draft captured. A POST /api/technicians/appointments endpoint is not yet available — refresh to re-sync server data.",
   532→              });
   533→              setScheduleOpen(false);
   534→            }}
   535→          />
   536→        </DialogContent>
   537→      </Dialog>
   538→    </Card>
   539→  );
   540→}

   798→// ---------------------------------------------------------------------------
   799→// Record Measurement dialog — fetches schemas + sources + profiles, POSTs
   800→// ---------------------------------------------------------------------------
   801→
   802→function RecordMeasurementDialog({
   803→  open,
   804→  onOpenChange,
   805→  onRecorded,
   806→}: {
   807→  open: boolean;
   808→  onOpenChange: (v: boolean) => void;
   809→  onRecorded: () => void;
   810→}) {
   811→  const [schemas, setSchemas] = useState<HealthSchema[]>([]);
   812→  const [sources, setSources] = useState<HealthSource[]>([]);
   813→  const [profiles, setProfiles] = useState<HealthProfile[]>([]);
   814→  const [loadingMeta, setLoadingMeta] = useState(false);
   815→  const [metaError, setMetaError] = useState<string | null>(null);
   816→
   817→  const [schemaId, setSchemaId] = useState("");
   818→  const [sourceId, setSourceId] = useState("");
   819→  const [profileId, setProfileId] = useState("");
   820→  const [unitId, setUnitId] = useState("");
   821→  const [value, setValue] = useState("");
   822→  const [tags, setTags] = useState("technician-recorded");
   823→  const [submitting, setSubmitting] = useState(false);
   824→
   825→  // Load metadata the first time the dialog opens.
   826→  useEffect(() => {
   827→    if (!open) return;
   828→    let cancelled = false;
   829→    setLoadingMeta(true);
   830→    setMetaError(null);
   831→    (async () => {
   832→      try {
   833→        const [schemaRes, sourceRes, profileRes] = await Promise.allSettled([
   834→          fetch("/api/health/schemas", { cache: "no-store" }),
   835→          fetch("/api/health/sources", { cache: "no-store" }),
   836→          fetch("/api/health/profiles", { cache: "no-store" }),
   837→        ]);
   838→
   839→        const readJson = async <T,>(r: PromiseSettledResult<Response>, fallback: T): Promise<T> => {
   840→          if (r.status !== "fulfilled") return fallback;
   841→          try {
   842→            const j = (await r.value.json()) as { ok?: boolean; data?: T };
   843→            if (!j?.ok) return fallback;
   844→            return (j.data as T) ?? fallback;
   845→          } catch {
   846→            return fallback;
   847→          }
   848→        };
   849→
   850→        const schemaData = await readJson<HealthSchema[] | { schemas?: HealthSchema[] }>(schemaRes, []);
   851→        const sourceData = await readJson<{ sources?: HealthSource[] } | HealthSource[]>(sourceRes, { sources: [] });
   852→        const profileData = await readJson<HealthProfile[] | { profiles?: HealthProfile[] }>(profileRes, []);
   853→
   854→        const schemasArr = Array.isArray(schemaData) ? schemaData : [];
   855→        const sourcesArr = Array.isArray(sourceData) ? sourceData : sourceData.sources ?? [];
   856→        const profilesArr = Array.isArray(profileData) ? profileData : [];
   857→
   858→        if (cancelled) return;
   859→        setSchemas(schemasArr);
   860→        setSources(sourcesArr);
   861→        setProfiles(profilesArr);
   862→
   863→        // Defaults — pick first option of each so the form is submittable immediately.
   864→        if (schemasArr[0] && !schemaId) {
   865→          setSchemaId(schemasArr[0].id);
   866→          const u = schemasArr[0].defaultUnit ?? schemasArr[0].allowedUnits?.[0] ?? "";
   867→          if (u) setUnitId(u);
   868→        }
   869→        if (sourcesArr[0] && !sourceId) setSourceId(sourcesArr[0].id);
   870→        if (profilesArr[0] && !profileId) setProfileId(profilesArr[0].id);
   871→
   872→        if (schemasArr.length === 0) {
   873→          setMetaError("No measurement schemas available. Ensure the health platform is seeded.");
   874→        }
   875→      } catch (err) {
   876→        if (!cancelled) {
   877→          const msg = err instanceof Error ? err.message : "Could not load measurement metadata";
   878→          setMetaError(msg);
   879→          toast({ title: "Load failed", description: msg, variant: "destructive" });
   880→        }
   881→      } finally {
   882→        if (!cancelled) setLoadingMeta(false);
   883→      }
   884→    })();
   885→    return () => { cancelled = true; };
   886→  }, [open]);
   887→
   888→  // When schema changes, sync the unit to the schema's default / first allowed unit.
   889→  const onSchemaChange = (id: string) => {
   890→    setSchemaId(id);
   891→    const s = schemas.find((x) => x.id === id);
   892→    const u = s?.defaultUnit ?? s?.allowedUnits?.[0] ?? "";
   893→    if (u) setUnitId(u);
   894→  };
   895→
   896→  const submit = async () => {
   897→    if (!schemaId || !profileId || value === "" || !unitId) {
   898→      toast({
   899→        title: "Missing fields",
   900→        description: "Schema, profile, value and unit are required.",
   901→        variant: "destructive",
   902→      });
   903→      return;
   904→    }
   905→    setSubmitting(true);
   906→    try {
   907→      const parsed = Number(value);
   908→      if (Number.isNaN(parsed)) {
   909→        toast({ title: "Invalid value", description: "Value must be a number.", variant: "destructive" });
   910→        setSubmitting(false);
   911→        return;
   912→      }
   913→      const body = {
   914→        schemaId,
   915→        profileId,
   916→        value: parsed,
   917→        unitId,
   918→        sourceId: sourceId || undefined,
   919→        collectedBy: "technician",
   920→        tags: tags
   921→          .split(",")
   922→          .map((t) => t.trim())
   923→          .filter(Boolean),
   924→      };
   925→      const res = await fetch("/api/health/measurements", {
   926→        method: "POST",
   927→        headers: { "Content-Type": "application/json" },
   928→        body: JSON.stringify(body),
   929→      });
   930→      const data = (await res.json()) as {
   931→        ok?: boolean;
   932→        data?: { measurementId?: string; verificationState?: string };
   933→        error?: { message?: string };
   934→      };
   935→      if (data.ok) {
   936→        toast({
   937→          title: "Measurement recorded",
   938→          description: `Verification state: ${data.data?.verificationState ?? "pending"}`,
   939→        });
   940→        setValue("");
   941→        onRecorded();
   942→      } else {
   943→        toast({
   944→          title: "Recording failed",
   945→          description: data.error?.message ?? "Server rejected the measurement",
   946→          variant: "destructive",
   947→        });
   948→      }
   949→    } catch (err) {
   950→      toast({
   951→        title: "Network error",
   952→        description: err instanceof Error ? err.message : "Could not reach server",
   953→        variant: "destructive",
   954→      });
   955→    } finally {
   956→      setSubmitting(false);
   957→    }
   958→  };
   959→
   960→  const selectedSchema = schemas.find((s) => s.id === schemaId);
   961→  const allowedUnits = selectedSchema?.allowedUnits ?? [];
   962→
   963→  return (
   964→    <Dialog open={open} onOpenChange={onOpenChange}>
   965→      <DialogContent className="sm:max-w-md">
   966→        <DialogHeader>
   967→          <DialogTitle>Record Measurement</DialogTitle>
   968→          <DialogDescription>
   969→            Submit a new measurement against a participant&apos;s health profile. The platform
   970→            validates it against the schema and assigns a verification state.
   971→          </DialogDescription>
   972→        </DialogHeader>
   973→
   974→        {loadingMeta ? (
   975→          <div className="py-6 text-center text-sm text-muted-foreground flex items-center justify-center gap-2">
   976→            <RefreshCw className="h-4 w-4 animate-spin" /> Loading measurement metadata…
   977→          </div>
   978→        ) : metaError ? (
   979→          <div className="rounded-md border border-destructive/40 bg-destructive/5 p-3 text-xs text-destructive">
   980→            {metaError}
   981→          </div>
   982→        ) : (
   983→          <div className="space-y-3 py-1">
   984→            <div className="space-y-1.5">
   985→              <Label>Measurement Type</Label>
   986→              <Select value={schemaId} onValueChange={onSchemaChange}>
   987→                <SelectTrigger><SelectValue placeholder="Select type" /></SelectTrigger>
   988→                <SelectContent className="max-h-72">
   989→                  {schemas.map((s) => (
   990→                    <SelectItem key={s.id} value={s.id}>
   991→                      {s.name}
   992→                      {s.category ? ` · ${s.category}` : ""}
   993→                    </SelectItem>
   994→                  ))}
   995→                </SelectContent>
   996→              </Select>
   997→            </div>
   998→
   999→            <div className="grid grid-cols-2 gap-3">
  1000→              <div className="space-y-1.5">
  1001→                <Label>Unit</Label>
  1002→                {allowedUnits.length > 0 ? (
  1003→                  <Select value={unitId} onValueChange={setUnitId}>
  1004→                    <SelectTrigger><SelectValue placeholder="Select unit" /></SelectTrigger>
  1005→                    <SelectContent>
  1006→                      {allowedUnits.map((u) => (
  1007→                        <SelectItem key={u} value={u}>{u}</SelectItem>
  1008→                      ))}
  1009→                    </SelectContent>
  1010→                  </Select>
  1011→                ) : (
  1012→                  <Input
  1013→                    value={unitId}
  1014→                    onChange={(e) => setUnitId(e.target.value)}
  1015→                    placeholder="unit id"
  1016→                  />
  1017→                )}
  1018→              </div>
  1019→              <div className="space-y-1.5">
  1020→                <Label htmlFor="meas-value">Value</Label>
  1021→                <Input
  1022→                  id="meas-value"
  1023→                  type="number"
  1024→                  step="any"
  1025→                  value={value}
  1026→                  onChange={(e) => setValue(e.target.value)}
  1027→                  placeholder="0.0"
  1028→                />
  1029→              </div>

  1031→
  1032→            <div className="space-y-1.5">

  1034→              <Select value={sourceId} onValueChange={setSourceId}>
  1035→                <SelectTrigger><SelectValue placeholder="Select source" /></SelectTrigger>
  1036→                <SelectContent className="max-h-72">

  1038→                    <SelectItem key={s.id} value={s.id}>

  1042→                  ))}

  1046→
  1047→            <div className="space-y-1.5">
  1048→              <Label>Participant Profile</Label>
  1049→              {profiles.length > 0 ? (
  1050→                <Select value={profileId} onValueChange={setProfileId}>

  1066→                />

  1069→
  1070→            <div className="space-y-1.5">

  1214→function fmtDate(iso: string | undefined | null, short = false): string {
  1215→  if (!iso) return "—";
  1216→  const d = new Date(iso);
  1217→  if (Number.isNaN(d.getTime())) return iso;
  1218→  if (short) {
  1219→    return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
  1220→  }
  1221→  return d.toLocaleString(undefined, {
  1222→    year: "numeric",
  1223→    month: "short",
  1224→    day: "numeric",
  1225→    hour: "2-digit",
  1226→    minute: "2-digit",
  1227→  });
  1228→}
