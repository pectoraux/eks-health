   239→export function OrgAdminDashboard({ data, onRefresh }: { data: DashboardData; onRefresh: () => void }) {
   240→  // Sub-API state — loaded client-side; the dashboard route only ships
   241→  // population.stats (analytics queries), everything else is fetched here.
   242→  const [identityOrgs, setIdentityOrgs] = useState<IdentityOrg[]>([]);
   243→  const [popOrgs, setPopOrgs] = useState<PopulationOrg[]>([]);
   244→  const [popStats, setPopStats] = useState<PopulationStats>({});
   245→  const [membershipStats, setMembershipStats] = useState<MembershipStats>({});
   246→  const [fundingStats, setFundingStats] = useState<FundingStats>({});
   247→  const [campaigns, setCampaigns] = useState<CampaignListItem[]>([]);
   248→  const [campaignStats, setCampaignStats] = useState<CampaignStats>({});
   249→  const [accounts, setAccounts] = useState<Account[]>([]);
   250→
   251→  const [loadingSub, setLoadingSub] = useState(true);
   252→  const [subError, setSubError] = useState<string | null>(null);
   253→
   254→  // Per-org expand + detail cache.
   255→  const [openOrgs, setOpenOrgs] = useState<Set<string>>(new Set());
   256→  const [orgDetail, setOrgDetail] = useState<Record<string, OrgDetail | null>>({});
   257→  const [orgDetailLoading, setOrgDetailLoading] = useState<Record<string, boolean>>({});
   258→
   259→  // Dialogs.
   260→  const [createOpen, setCreateOpen] = useState(false);
   261→  const [addMemberOrg, setAddMemberOrg] = useState<IdentityOrg | null>(null);
   262→  const [inviteMemberOrg, setInviteMemberOrg] = useState<IdentityOrg | null>(null);
   263→
   264→  // --- Sub-data load -----------------------------------------------------
   265→
   266→  const loadSubData = useCallback(async () => {
   267→    setLoadingSub(true);
   268→    setSubError(null);
   269→    try {
   270→      const results = await Promise.allSettled([
   271→        fetch("/api/identity/orgs", { cache: "no-store" }),
   272→        fetch("/api/population/organizations", { cache: "no-store" }),
   273→        fetch("/api/population/memberships", { cache: "no-store" }),
   274→        fetch("/api/population/funding", { cache: "no-store" }),
   275→        fetch("/api/population/campaigns", { cache: "no-store" }),
   276→        fetch("/api/identity/accounts", { cache: "no-store" }),
   277→      ]);
   278→
   279→      const readJson = async <T,>(r: PromiseSettledResult<Response>, fallback: T, label: string): Promise<T> => {
   280→        if (r.status !== "fulfilled") {
   281→          console.warn(`[org-admin-dashboard] ${label} request rejected`);
   282→          return fallback;
   283→        }
   284→        try {
   285→          const j = (await r.value.json()) as { ok?: boolean; data?: T; error?: { message?: string } };
   286→          if (!j?.ok) {
   287→            console.warn(`[org-admin-dashboard] ${label} returned ok=false`, j?.error?.message);
   288→            return fallback;
   289→          }
   290→          return (j.data as T) ?? fallback;
   291→        } catch {
   292→          return fallback;
   293→        }
   294→      };
   295→
   296→      const [orgs, pop, mem, fund, camps, accts] = await Promise.all([
   297→        readJson<IdentityOrg[]>(results[0], [], "identity/orgs"),
   298→        readJson<{ organizations: PopulationOrg[]; stats: PopulationStats }>(results[1], { organizations: [], stats: {} }, "population/organizations"),
   299→        readJson<{ stats: MembershipStats }>(results[2], { stats: {} }, "population/memberships"),
   300→        readJson<{ stats: FundingStats }>(results[3], { stats: {} }, "population/funding"),
   301→        readJson<{ campaigns: CampaignListItem[]; stats: CampaignStats }>(results[4], { campaigns: [], stats: {} }, "population/campaigns"),
   302→        readJson<Account[]>(results[5], [], "identity/accounts"),
   303→      ]);
   304→
   305→      setIdentityOrgs(orgs);
   306→      setPopOrgs(pop.organizations ?? []);
   307→      setPopStats(pop.stats ?? {});
   308→      setMembershipStats(mem.stats ?? {});
   309→      setFundingStats(fund.stats ?? {});
   310→      setCampaigns(camps.campaigns ?? []);
   311→      setCampaignStats(camps.stats ?? {});
   312→      setAccounts(accts);
   313→    } catch (err) {
   314→      const msg = err instanceof Error ? err.message : "Failed to load org admin data";
   315→      setSubError(msg);
   316→      toast({ title: "Load failed", description: msg, variant: "destructive" });
   317→    } finally {
   318→      setLoadingSub(false);
   319→    }
   320→  }, []);
   321→
   322→  useEffect(() => {
   323→    void loadSubData();
   324→  }, [loadSubData]);
   325→
   326→  // --- Per-org detail fetch (lazy on expand) -----------------------------
   327→
   328→  const fetchOrgDetail = useCallback(async (orgId: string) => {
   329→    if (orgDetail[orgId] !== undefined || orgDetailLoading[orgId]) return;
   330→    setOrgDetailLoading((s) => ({ ...s, [orgId]: true }));
   331→    try {
   332→      const res = await fetch(`/api/identity/orgs/${encodeURIComponent(orgId)}`, { cache: "no-store" });
   333→      const j = (await res.json()) as { ok?: boolean; data?: OrgDetail; error?: { message?: string } };
   334→      if (j.ok && j.data) {
   335→        setOrgDetail((s) => ({ ...s, [orgId]: j.data as OrgDetail }));
   336→      } else {
   337→        setOrgDetail((s) => ({ ...s, [orgId]: null }));
   338→        console.warn(`[org-admin-dashboard] org detail ${orgId} returned ok=false`, j.error?.message);
   339→      }
   340→    } catch (err) {
   341→      setOrgDetail((s) => ({ ...s, [orgId]: null }));
   342→      console.warn(`[org-admin-dashboard] org detail ${orgId} fetch failed`, err);
   343→    } finally {
   344→      setOrgDetailLoading((s) => ({ ...s, [orgId]: false }));
   345→    }
   346→  }, [orgDetail, orgDetailLoading]);
   347→
   348→  const toggleOrg = useCallback((orgId: string) => {
   349→    setOpenOrgs((prev) => {
   350→      const next = new Set(prev);
   351→      if (next.has(orgId)) {
   352→        next.delete(orgId);
   353→      } else {
   354→        next.add(orgId);
   355→        void fetchOrgDetail(orgId);
   356→      }
   357→      return next;
   358→    });
   359→  }, [fetchOrgDetail]);
   360→
   361→  // --- Actions -----------------------------------------------------------
   362→
   363→  const refreshOrgDetail = useCallback(async (orgId: string) => {
   364→    setOrgDetail((s) => ({ ...s, [orgId]: undefined }));
   365→    setOrgDetailLoading((s) => ({ ...s, [orgId]: true }));
   366→    try {
   367→      const res = await fetch(`/api/identity/orgs/${encodeURIComponent(orgId)}`, { cache: "no-store" });
   368→      const j = (await res.json()) as { ok?: boolean; data?: OrgDetail; error?: { message?: string } };
   369→      if (j.ok && j.data) {
   370→        setOrgDetail((s) => ({ ...s, [orgId]: j.data as OrgDetail }));
   371→      } else {
   372→        setOrgDetail((s) => ({ ...s, [orgId]: null }));
   373→      }
   374→    } catch {
   375→      setOrgDetail((s) => ({ ...s, [orgId]: null }));
   376→    } finally {
   377→      setOrgDetailLoading((s) => ({ ...s, [orgId]: false }));
   378→    }
   379→  }, []);
   380→
   381→  const handleOrgCreated = useCallback(() => {
   382→    onRefresh();
   383→    void loadSubData();
   384→  }, [onRefresh, loadSubData]);
   385→
   386→  const handleMemberAdded = useCallback((orgId: string) => {
   387→    void refreshOrgDetail(orgId);
   388→    onRefresh();
   389→    void loadSubData();
   390→  }, [refreshOrgDetail, onRefresh, loadSubData]);
   391→
   392→  const handleManualRefresh = useCallback(() => {
   393→    onRefresh();
   394→    // Invalidate cached org details so re-expand refetches.
   395→    setOrgDetail({});
   396→    void loadSubData();
   397→  }, [onRefresh, loadSubData]);
   398→
   399→  // --- Derived -----------------------------------------------------------
   400→
   401→  // Build a slug → population org lookup so identity orgs can pick up
   402→  // memberCount / tier / country from the population subsystem.
   403→  const popBySlug = useMemo(() => {
   404→    const m = new Map<string, PopulationOrg>();
   405→    for (const p of popOrgs) m.set(p.slug, p);
   406→    return m;
   407→  }, [popOrgs]);
   408→
   409→  // Compute hierarchy depth for each identity org by walking parentId.
   410→  const depthMap = useMemo(() => {
   411→    const byId = new Map<string, IdentityOrg>();
   412→    for (const o of identityOrgs) byId.set(o.id, o);
   413→    const cache = new Map<string, number>();
   414→    const depthOf = (id: string): number => {
   415→      if (cache.has(id)) return cache.get(id) as number;
   416→      const o = byId.get(id);
   417→      if (!o || !o.parentId || !byId.has(o.parentId)) {
   418→        cache.set(id, 0);
   419→        return 0;
   420→      }
   421→      const d = 1 + depthOf(o.parentId);
   422→      cache.set(id, d);
   423→      return d;
   424→    };
   425→    const m = new Map<string, number>();
   426→    for (const o of identityOrgs) m.set(o.id, depthOf(o.id));
   427→    return m;
   428→  }, [identityOrgs]);
   429→
   430→  const mergedOrgs: MergedOrg[] = useMemo(() => {
   431→    return identityOrgs.map((org) => {
   432→      const pop = popBySlug.get(org.slug);
   433→      return {
   434→        org,
   435→        memberCount: pop?.memberCount ?? 0,
   436→        activeMemberCount: pop?.activeMemberCount ?? 0,
   437→        tier: pop?.tier ?? "free",
   438→        country: pop?.country ?? "",
   439→        popStatus: pop?.status,
   440→        depth: depthMap.get(org.id) ?? 0,
   441→      };
   442→    });
   443→  }, [identityOrgs, popBySlug, depthMap]);
   444→
   445→  const totalOrgs = popStats.total ?? identityOrgs.length;
   446→  const totalMembers = membershipStats.total ?? popStats.totalMembers ?? 0;
   447→  const activeCampaigns = campaignStats.byStatus?.active ?? campaigns.filter((c) => c.status === "active").length;
   448→  const populationReach = data.population?.stats?.totalQueries ?? data.population?.stats?.totalContexts ?? 0;
   449→
   450→  return (
   451→    <div className="space-y-6">
   452→      {/* Top stat row + Refresh */}
   453→      <div className="flex flex-col gap-3 sm:flex-row sm:items-stretch">
   454→        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 flex-1">
   455→          <StatCard
   456→            icon={<Building2 className="h-4 w-4" />}
   457→            label="Organizations"
   458→            value={totalOrgs}
   459→            hint={`${popStats.active ?? identityOrgs.filter((o) => o.status === "active").length} active`}
   460→            loading={loadingSub && !identityOrgs.length}
   461→            accent
   462→          />
   463→          <StatCard
   464→            icon={<Users className="h-4 w-4" />}
   465→            label="Total Members"
   466→            value={totalMembers}
   467→            hint={`${membershipStats.active ?? 0} active · ${membershipStats.invited ?? 0} invited`}
   468→            loading={loadingSub}
   469→          />
   470→          <StatCard
   471→            icon={<Trophy className="h-4 w-4" />}
   472→            label="Active Campaigns"
   473→            value={activeCampaigns}
   474→            hint={`${campaignStats.total ?? campaigns.length} total`}
   475→            loading={loadingSub}
   476→          />
   477→          <StatCard
   478→            icon={<Activity className="h-4 w-4" />}
   479→            label="Population Reach"
   480→            value={populationReach}
   481→            hint={data.population?.stats?.avgGoals ? `${data.population.stats.avgGoals} avg goals` : "analytics queries"}
   482→            loading={loadingSub}
   483→          />
   484→        </div>
   485→        <div className="flex sm:flex-col gap-2 sm:justify-end">
   486→          <Button variant="outline" onClick={handleManualRefresh} className="h-9" disabled={loadingSub}>
   487→            <RefreshCw className={`h-4 w-4 mr-1 ${loadingSub ? "animate-spin" : ""}`} />
   488→            Refresh
   489→          </Button>
   490→        </div>
   491→      </div>
   492→
   493→      {subError && (
   494→        <div className="rounded-md border border-destructive/40 bg-destructive/5 p-3 text-xs text-destructive">
   495→          Could not load some org admin data: {subError}. Showing partial view.
   496→        </div>
   497→      )}
   498→
   499→      {/* Action bar */}
   500→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   501→        <div>
   502→          <h2 className="text-lg font-semibold tracking-tight">Organization Management</h2>
   503→          <p className="text-sm text-muted-foreground">
   504→            Create, expand, and administer your organizations, members, and teams.
   505→          </p>
   506→        </div>
   507→        <Button onClick={() => setCreateOpen(true)} className="h-9">
   508→          <Plus className="h-4 w-4 mr-1" />
   509→          Create Organization
   510→        </Button>
   511→      </div>
   512→
   513→      {/* Organizations card (full width, expandable) */}
   514→      <OrganizationsCard
   515→        orgs={mergedOrgs}
   516→        openOrgs={openOrgs}
   517→        onToggle={toggleOrg}
   518→        orgDetail={orgDetail}
   519→        orgDetailLoading={orgDetailLoading}
   520→        loading={loadingSub}
   521→        accounts={accounts}
   522→        onAddMember={(org) => setAddMemberOrg(org)}
   523→        onInviteMember={(org) => setInviteMemberOrg(org)}
   524→        onMemberAdded={handleMemberAdded}
   525→      />
   526→
   527→      {/* Active Campaigns + Funding Overview */}
   528→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   529→        <CampaignsCard campaigns={campaigns} stats={campaignStats} loading={loadingSub} />
   530→        <FundingOverviewCard stats={fundingStats} loading={loadingSub} />
   531→      </div>
   532→
   533→      {/* Population Analytics (full width) */}
   534→      <PopulationAnalyticsCard
   535→        stats={data.population?.stats ?? {}}
   536→        membershipStats={membershipStats}
   537→        campaignStats={campaignStats}
   538→        popStats={popStats}
   539→        loading={loadingSub}
   540→      />
   541→
   542→      {/* Dialogs */}
   543→      <CreateOrgDialog
   544→        open={createOpen}
   545→        onOpenChange={setCreateOpen}
   546→        onCreated={handleOrgCreated}
   547→      />
   548→      <AddMemberDialog
   549→        org={addMemberOrg}
   550→        accounts={accounts}
   551→        onOpenChange={(o) => !o && setAddMemberOrg(null)}
   552→        onAdded={handleMemberAdded}
   553→      />
   554→      <InviteMemberDialog
   555→        org={inviteMemberOrg}
   556→        onOpenChange={(o) => !o && setInviteMemberOrg(null)}
   557→        onInvited={() => { onRefresh(); void loadSubData(); }}
   558→      />
   559→    </div>
   560→  );
   561→}
