     1→#!/usr/bin/env bash
     2→# audit-systematic.sh — SYSTEMATIC bug audit of live Eks-Health app
     3→#
     4→# Site:    https://eks-health.vercel.app  (Vercel production, serverless)
     5→# Task ID: audit-systematic-1
     6→#
     7→# Goal:
     8→#   Drive a real Chrome session via agent-browser through every page and
     9→#   navigation flow. For each test, capture PASS/FAIL + observed vs expected
    10→#   + console errors + final URL. The app uses in-memory state that does
    11→#   not persist across serverless cold starts, so we test auth/session
    12→#   persistence especially hard.
    13→#
    14→# Key design choices:
    15→#   - api_probe returns pipe-delimited strings "status|ok|snippet" — avoids
    16→#     JSON-string-escaping issues when grep'ing for ok:true.
    17→#   - All `wait N` calls are followed by `sleep` for reliability.
    18→#   - Element refs are extracted with EXACT patterns (e.g. 'button "Marketplace"')
    19→#     rather than loose alternations that match the first line containing
    20→#     "button".
    21→#   - Per-test session setup: we sign in fresh where needed so tests don't
    22→#     inherit unexpected state from prior tests.
    23→#
    24→# Artifacts:
    25→#   Logs:        /home/z/my-project/.zscripts/logs/audit-systematic/
    26→#   Screenshots: /home/z/my-project/.zscripts/screenshots/audit-systematic/
    27→#   Report:      /home/z/my-project/.zscripts/logs/audit-systematic/REPORT.txt
    28→#   Bugs:        /home/z/my-project/.zscripts/logs/audit-systematic/BUGS.txt
    29→
    30→set -uo pipefail
    31→
    32→BASE="https://eks-health.vercel.app"
    33→SESSION="eks-audit-sys"
    34→ROOT="/home/z/my-project/.zscripts"
    35→LOGS="${ROOT}/logs/audit-systematic"
    36→SHOTS="${ROOT}/screenshots/audit-systematic"
    37→REPORT="${LOGS}/REPORT.txt"
    38→BUGS="${LOGS}/BUGS.txt"
    39→mkdir -p "$LOGS" "$SHOTS"
    40→: >"$REPORT"
    41→: >"$BUGS"
    42→
    43→# agent-browser helper (isolated session).
    44→ab() { agent-browser --session "$SESSION" "$@"; }
    45→
    46→# Write a line to the report (and stdout).
    47→rpt() { printf '%s\n' "$*" | tee -a "$REPORT" >&2; }
    48→section() {
    49→  local bar="================================================================"
    50→  {
    51→    printf '\n%s\n%s\n%s\n' "$bar" "$*" "$bar"
    52→  } | tee -a "$REPORT" >&2
    53→}
    54→bug() { printf '%s\n' "$*" | tee -a "$BUGS" >&2; }
    55→
    56→# Capture console + page errors + interactive snapshot + screenshot.
    57→capture_diagnostics() {
    58→  local tag="$1"
    59→  ab console >"${LOGS}/${tag}.console.txt" 2>&1 || true
    60→  ab errors  >"${LOGS}/${tag}.errors.txt" 2>&1 || true
    61→  ab snapshot -i -c >"${LOGS}/${tag}.snapshot.txt" 2>&1 || true
    62→  ab screenshot "${SHOTS}/${tag}.png" >/dev/null 2>&1 || true
    63→}
    64→
    65→# Count console error-level messages for a tag.
    66→console_err_count() {
    67→  local f="${LOGS}/$1.console.txt"
    68→  if [ -s "$f" ]; then
    69→    grep -iE '"type":"error"|level=error|SEVERE|Error:|Failed to|Uncaught' "$f" 2>/dev/null | wc -l | tr -d ' '
    70→  else
    71→    echo 0
    72→  fi
    73→}
    74→
    75→# Run a fetch() inside the browser. Returns a pipe-delimited string:
    76→#   "<status>|<ok 0/1>|<snippet>"
    77→# The snippet is the first ~150 chars of either d.error.message or
    78→# JSON.stringify(d.data ?? d), with pipe characters replaced so it doesn't
    79→# break parsing.
    80→api_probe() {
    81→  local method="$1" path="$2" body="${3:-}"
    82→  local js
    83→  if [ -n "$body" ]; then
    84→    js="fetch('${path}',{method:'${method}',headers:{'Content-Type':'application/json'},body:JSON.stringify(${body}),cache:'no-store'}).then(async r=>{const d=await r.json().catch(()=>({}));const snip=(d.error?.message||JSON.stringify(d.data??d)||'').slice(0,180).replace(/\\|/g,'/');return [r.status,d.ok?1:0,snip].join('|')})"
    85→  else
    86→    js="fetch('${path}',{cache:'no-store'}).then(async r=>{const d=await r.json().catch(()=>({}));const snip=(d.error?.message||JSON.stringify(d.data??d)||'').slice(0,180).replace(/\\|/g,'/');return [r.status,d.ok?1:0,snip].join('|')})"
    87→  fi
    88→  ab eval "$js" 2>/dev/null | tail -1 | tr -d '"'
    89→}
    90→
    91→# Sign in via direct API (reliable). Returns "status|ok|email|persona|err".
    92→api_sign_in() {
    93→  local email="$1" pass="$2"
    94→  ab eval "fetch('/api/auth/sign-in',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({email:'${email}',password:'${pass}'})}).then(async r=>{const d=await r.json();return [r.status,d.ok?1:0,d.data?.email||'',d.data?.activePersona||'',d.error?.message||''].join('|')})" 2>/dev/null | tail -1 | tr -d '"'
    95→}
    96→
    97→# Sign in via the actual UI form (fills inputs + clicks Sign In button).
    98→# Sets up the dashboard state. Returns 0 on success.
    99→ui_sign_in() {
   100→  local email="$1" pass="$2"
   101→  ab open "${BASE}/sign-in" >/dev/null 2>&1
   102→  ab wait --load networkidle >/dev/null 2>&1
   103→  sleep 1
   104→  ab snapshot -i -c >"${LOGS}/.signin.snap" 2>&1
   105→  local emailref passref signinref
   106→  emailref="$(grep -iE 'textbox.*Email' "${LOGS}/.signin.snap" | head -1 | sed -E 's/.*\[ref=([a-z0-9]+)\].*/\1/')"
   107→  passref="$(grep -iE 'textbox.*Password' "${LOGS}/.signin.snap" | head -1 | sed -E 's/.*\[ref=([a-z0-9]+)\].*/\1/')"
   108→  signinref="$(grep -E 'button "Sign In"' "${LOGS}/.signin.snap" | head -1 | sed -E 's/.*\[ref=([a-z0-9]+)\].*/\1/')"
   109→  if [ -z "$emailref" ] || [ -z "$passref" ] || [ -z "$signinref" ]; then
   110→    echo "ERROR: could not locate sign-in form fields (email=$emailref pass=$passref btn=$signinref)" >&2
   111→    return 1
   112→  fi
   113→  ab fill "@${emailref}" "$email" >/dev/null 2>&1 || true
   114→  ab fill "@${passref}"  "$pass"  >/dev/null 2>&1 || true
   115→  ab click "@${signinref}" >/dev/null 2>&1 || true
   116→  ab wait --url "**/dashboard" --timeout 15000 >/dev/null 2>&1 || true
   117→  sleep 3
   118→  return 0
   119→}
   120→
   121→# Extract an element ref by EXACTLY matching a snapshot pattern like
   122→# 'button "Marketplace"' (so we don't grab an unrelated button line).
   123→# Args: <snapshot-file> <pattern>
   124→ref_for() {
   125→  grep -E "$2" "$1" 2>/dev/null | head -1 | sed -E 's/.*\[ref=([a-z0-9]+)\].*/\1/'
   126→}

   129→marketplace_card_count() {
   130→  ab eval '[...document.querySelectorAll("main .grid > div")].filter(c=>/installs/i.test(c.textContent||"")).length' 2>/dev/null | tail -1 | tr -d '"'
   131→}

   133→# Get "name|slug" of the first marketplace listing card.
   134→first_listing_slug() {
   135→  ab eval '(function(){const c=document.querySelector("main .grid > div");if(!c)return "none|none";const h3=c.querySelector("h3");const name=h3?h3.textContent.trim():"?";const slug=name.toLowerCase().replace(/[^a-z0-9]+/g,"-").replace(/^-|-$/g,"");return name+"|"+slug})()' 2>/dev/null | tail -1 | tr -d '"'
   136→}

   138→# Pass/Fail counters
   139→PASS_COUNT=0
   140→FAIL_COUNT=0
   141→WARN_COUNT=0
   142→declare -a RESULTS=()
   143→
   144→record() {
   145→  # record <test_id> <PASS|FAIL|WARN> <summary>
   146→  RESULTS+=("$1|$2|$3")
   147→  case "$2" in
   148→    PASS) PASS_COUNT=$((PASS_COUNT+1));;
   149→    FAIL) FAIL_COUNT=$((FAIL_COUNT+1));;
   150→    WARN) WARN_COUNT=$((WARN_COUNT+1));;
   151→  esac
   152→  rpt "  RESULT: $2 — $3"
   153→}
   154→
   155→# Check that an api_probe result has status=200 and ok=1.
   156→# Args: <probe-result> <label>
   157→api_ok() {
   158→  local probe="$1" label="$2"
   159→  local status ok
   160→  status="$(echo "$probe" | cut -d'|' -f1)"
   161→  ok="$(echo "$probe" | cut -d'|' -f2)"
   162→  if [ "$status" = "200" ] && [ "$ok" = "1" ]; then
   163→    return 0
   164→  fi
   165→  return 1
   166→}
   167→
   168→# Stricter check: status=200, ok=1, AND the snippet contains real session
   169→# fields (accountId/email). /api/auth/session returns {ok:true,data:null} when
   170→# the cookie is invalid on a cold serverless instance — api_ok alone would
   171→# treat that as success, masking the real auth-persistence bug.
   172→# Args: <probe-result>
   173→session_is_valid() {
   174→  local probe="$1"
   175→  api_ok "$probe" session || return 1
   176→  # Look for real session fields. The snippet is JSON-stringified with escaped
   177→  # quotes (e.g. {\"accountId\":\"acc_...\"}), so grep for the field name
   178→  # without requiring exact quote matching.
   179→  echo "$probe" | grep -qiE 'accountId|email|displayName' || return 1
   180→  # Explicitly reject the null-session shape.
   181→  if echo "$probe" | grep -qiE 'data\\*":null|ok\\*":true,\\*"data\\*":null'; then
   182→    # Only reject if accountId is NOT present (null session has no accountId)
   183→    if ! echo "$probe" | grep -qi 'accountId'; then
   184→      return 1
   185→    fi
   186→  fi
   187→  return 0
   188→}
   189→
   190→# Extract a field from a probe result. Args: <probe> <field-index 1-based>.
   191→probe_field() { echo "$1" | cut -d'|' -f"$2"; }
